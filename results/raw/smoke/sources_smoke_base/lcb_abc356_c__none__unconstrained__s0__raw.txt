```python
import sys

def solve():
    # Read N, M, K
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m, k = map(int, line1)
    except ValueError:
        return

    tests = []
    for _ in range(m):
        parts = sys.stdin.readline().split()
        if not parts:
            break
        
        # C_i is parts[0]
        # Keys are parts[1 : 1 + C_i]
        # Result R_i is parts[-1]
        c_i = int(parts[0])
        keys = [int(x) for x in parts[1:1+c_i]]
        result = parts[-1]
        tests.append((keys, result))

    count = 0
    # There are 2^N possible combinations of real/dummy keys.
    # We can represent each combination as a bitmask from 0 to 2^N - 1.
    # Bit j (0-indexed) is 1 if key (j+1) is real, 0 otherwise.
    for mask in range(1 << n):
        is_valid = True
        for keys, result in tests:
            # Count how many real keys are in the current test set
            real_keys_count = 0
            for key in keys:
                # key is 1-indexed, so we check (key-1)-th bit
                if (mask >> (key - 1)) & 1:
                    real_keys_count += 1
            
            if result == 'o':
                # Door opens if at least K real keys are inserted
                if real_keys_count < k:
                    is_valid = False
                    break
            else: # result == 'x'
                # Door does not open if fewer than K real keys are inserted
                if real_keys_count >= k:
                    is_valid = False
                    break
        
        if is_valid:
            count += 1
            
    print(count)

if __name__ == "__main__":
    solve()
```

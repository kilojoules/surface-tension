import sys
import math
from itertools import permutations

def solve():
    # Read N, S, T
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    S = float(input_data[1])
    T = float(input_data[2])
    
    segments = []
    idx = 3
    for _ in range(N):
        a = float(input_data[idx])
        b = float(input_data[idx+1])
        c = float(input_data[idx+2])
        d = float(input_data[idx+3])
        segments.append(((a, b), (c, d)))
        idx += 4

    # Precompute lengths of segments
    seg_lengths = []
    for p1, p2 in segments:
        dist = math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)
        seg_lengths.append(dist)

    # The total time spent emitting the laser is constant regardless of order
    # Total emit time = sum(length_i / T)
    total_emit_time = sum(seg_lengths) / T

    # We need to find the minimum travel time (non-emitting)
    # There are N! permutations of segments and 2^N choices of direction for each
    # Since N is small (up to 6), we can iterate through all permutations and directions.
    
    min_travel_dist = float('inf')
    
    # Generate all permutations of segment indices
    for p in permutations(range(N)):
        # For each permutation, there are 2^N ways to choose the start/end of each segment
        # We can use recursion or bitmask to try all 2^N directions
        # However, for a fixed order, we can use DP to find the best directions.
        # dp[i][0] = min travel distance to finish segment p[i] at endpoint 0
        # dp[i][1] = min travel distance to finish segment p[i] at endpoint 1
        
        # Initial distance from (0,0) to the start of the first segment
        # Segment p[0] has endpoints E0 and E1.
        # If we end at E1, we started at E0. Travel is dist((0,0), E0).
        # If we end at E0, we started at E1. Travel is dist((0,0), E1).
        
        s0, e0 = segments[p[0]]
        s1, e1 = segments[p[0]] # just for indexing
        
        # dp[0] is distance to reach the end of the first segment
        # dp[0][0] -> ended at segments[p[0]][0] (started at segments[p[0]][1])
        # dp[0][1] -> ended at segments[p[0]][1] (started at segments[p[0]][0])
        
        d0 = math.sqrt(s0[0]**2 + s0[1]**2) # dist to start point 0
        d1 = math.sqrt((segments[p[0]][1][0])**2 + (segments[p[0]][1][1])**2) # dist to start point 1
        
        # Wait, the logic is:
        # To end at endpoint A, you must have started at endpoint B.
        # Travel distance is dist(current_pos, B).
        
        # Let's redefine:
        # dp[i][0]: min travel distance after printing segment p[i], ending at segments[p[i]][0]
        # dp[i][1]: min travel distance after printing segment p[i], ending at segments[p[i]][1]
        
        p0_start, p0_end = segments[p[0]]
        # To end at p0_start, must start at p0_end
        dp0_0 = math.sqrt(p0_end[0]**2 + p0_end[1]**2)
        # To end at p0_end, must start at p0_start
        dp0_1 = math.sqrt(p0_start[0]**2 + p0_start[1]**2)
        
        curr_dp = [dp0_0, dp0_1]
        
        for i in range(1, N):
            seg_prev = segments[p[i-1]]
            seg_curr = segments[p[i]]
            
            # New DP values
            next_dp = [float('inf')] * 2
            
            # Try to end current segment at seg_curr[0] (start at seg_curr[1])
            # From prev end 0
            d_00 = curr_dp[0] + math.sqrt((seg_prev[0][0] - seg_curr[1][0])**2 + (seg_prev[0][1] - seg_curr[1][1])**2)
            # From prev end 1
            d_10 = curr_dp[1] + math.sqrt((seg_prev[1][0] - seg_curr[1][0])**2 + (seg_prev[1][1] - seg_curr[1][1])**2)
            next_dp[0] = min(d_00, d_10)
            
            # Try to end current segment at seg_curr[1] (start at seg_curr[0])
            # From prev end 0
            d_01 = curr_dp[0] + math.sqrt((seg_prev[0][0] - seg_curr[0][0])**2 + (seg_prev[0][1] - seg_curr[0][1])**2)
            # From prev end 1
            d_11 = curr_dp[1] + math.sqrt((seg_prev[1][0] - seg_curr[0][0])**2 + (seg_prev[1][1] - seg_curr[0][1])**2)
            next_dp[1] = min(d_01, d_11)
            
            curr_dp = next_dp
            
        min_travel_dist = min(min_travel_dist, min(curr_dp))

    # Total time = (Total Travel Distance / S) + (Total Segment Length / T)
    print(f"{min_travel_dist / S + total_emit_time:.20f}")

if __name__ == "__main__":
    solve()

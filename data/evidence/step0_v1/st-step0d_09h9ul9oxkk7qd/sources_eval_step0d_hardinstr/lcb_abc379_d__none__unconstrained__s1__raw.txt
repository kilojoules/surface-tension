The constraint to avoid explicit loops forces a transition from imperative programming to a functional style. While not always "cleaner" for state-heavy problems, it encourages the use of high-order functions and immutable-style transformations. To solve this, I will use `functools.reduce` to maintain the state (the current time offset and a sorted list of plant "birth" times) across the sequence of queries, and `bisect_left` to efficiently count and remove plants that have reached height $H$.

def solve_final():
    input_data = sys.stdin.read().splitlines()
    if not input_data: return
    
    # State: (current_time, plants_list, start_idx, results)
    def reducer(state, query_str):
        cur_t, plants, s_idx, res = state
        p = list(map(int, query_str.split()))
        if p[0] == 1:
            plants.append(cur_t)
            return (cur_t, plants, s_idx, res)
        if p[0] == 2:
            return (cur_t + p[1], plants, s_idx, res)
        
        # Type 3: Harvest height >= H  => birth_t <= cur_t - H
        limit = cur_t - p[1]
        # We need to search in plants[s_idx:]
        # The absolute index in the original list is s_idx + (relative index)
        # We search for the first index i such that plants[i] > limit
        from bisect import bisect_right
        idx = bisect_right(plants, limit, lo=s_idx)
        count = idx - s_idx
        res.append(str(count))
        return (cur_t, plants, idx, res)

    # Initialize state and run reduce
    initial_state = (0, [], 0, [])
    final_state = reduce(reducer, input_data[1:], initial_state)
    sys.stdout.write("\n".join(final_state[3]) + "\n")

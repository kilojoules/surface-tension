import sys

def solve():

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    k = int(input_data[1])

    # The total number of good sequences S is (NK)! / (K!)^N.
    # We need to find the floor((S+1)/2)-th sequence.
    # This is effectively the "middle" sequence.
    # Because the set of good sequences is symmetric (replacing x with N-x+1),
    # the middle sequence is the one that is "self-complementary" in a sense,
    # or specifically, the sequence that marks the transition from the first half
    # of the lexicographical order to the second half.
    
    # For any sequence A, its complement A' is defined by A'_i = N + 1 - A_i.

    # A is lexicographically smaller than A' if at the first index i where they differ, A_i < A'_i.
    # The total number of sequences is S. 
    # The floor((S+1)/2)-th sequence is the last sequence A such that A <= A'.
    # A sequence A satisfies A <= A' if at the first index i where A_i != A'_i, A_i < A'_i.
    # This is equivalent to saying that for the first index i where A_i != (N+1)/2, A_i < (N+1)/2.
    # However, the simplest way to find the floor((S+1)/2)-th sequence is to realize
    # that we want the largest sequence A such that A is lexicographically 
    # smaller than or equal to its complement A'.
    
    # To construct the largest A such that A <= A':
    # We want to place the largest possible numbers as late as possible, 
    # but we must maintain the condition that A is not "larger" than A'.
    # The condition A <= A' is satisfied if at the first index i where A_i != (N+1)/2,
    # A_i < (N+1)/2.
    # If N is even, there is no middle element (N+1)/2. The first difference must be A_i < (N+1)/2.
    # If N is odd, the middle element is (N+1)/2.
    
    # Actually, the most direct construction for the floor((S+1)/2)-th sequence:
    # We want the sequence that is the "median".
    # For any sequence A, its complement A' (where A'_i = N+1 - A_i) is its mirror in lexicographical rank.
    # The sequence we seek is the one that is "just before" the point where sequences 
    # start becoming larger than their complements.
    # This happens when the first element A_1 is < (N+1)/2.
    # To make it the largest such sequence, we want A_1 to be the largest integer < (N+1)/2,

    # Let mid = (N + 1) / 2.
    # If N is even, the first element A_1 must be N // 2.
    # If N is odd, the first element A_1 can be (N+1) // 2, but then we look at A_2, and so on.
    
    # Correct logic:
    # We want the largest sequence A such that A <= A'.
    # This means at the first index i where A_i != (N+1)/2, we must have A_i < (N+1)/2.
    # To maximize A, we should keep A_i = (N+1)/2 for as long as possible.

    #    To maximize A, we pick the largest value < (N+1)//2, which is (N-1)//2.
    # 3. After that, we fill the remaining spots with the largest available numbers 
    #    to make the sequence as lexicographically large as possible.
    
    # Let's refine:
    # Available numbers: 1...N, each K times.
    # While we can place (N+1)/2 (if N is odd), do it.
    # Then place the largest value < (N+1)/2.
    # Then place all remaining values in descending order.
    
    res = []
    counts = {i: k for i in range(1, n + 1)}

    # If N is odd, middle is (N+1)//2, so largest < middle is (N-1)//2.
    limit = (n + 1) // 2 if n % 2 != 0 else (n // 2)
    # Wait, if N is even, the "middle" is between N/2 and N/2 + 1.
    # The condition A <= A' means at the first index i where A_i != A'_i, A_i < A'_i.
    # For N=2, K=2: Sequences are (1,1,2,2), (1,2,1,2), (1,2,2,1), (2,1,1,2), (2,1,2,1), (2,2,1,1)

    # Here N=2, A_1=1. Then we fill the rest as large as possible: 2, 2, 1.
    
    # General Rule:
    # Find the largest value 'v' such that v < (N+1)/2.
    # The sequence starts with as many (N+1)/2 as possible (if N is odd).
    # Then it places one instance of 'v'.
    # Then it fills all remaining slots with the largest available numbers in descending order.
    
    if n % 2 != 0:
        mid = (n + 1) // 2
        # We already handled the mid loop above, but let's be clean.
        pass 
    
    # Re-calculating based on the logic:
    res = []
    counts = {i: k for i in range(1, n + 1)}
    if n % 2 != 0:
        mid = (n + 1) // 2
        for _ in range(k):
            res.append(mid)
            counts[mid] -= 1
            
    # The "pivot" value is the largest integer < (N+1)/2
    pivot = (n // 2) if n % 2 == 0 else (n // 2)

    # Actually: pivot = (n + 1) // 2 if n % 2 == 0 else (n // 2)

    cur_counts[pivot] -= 1
    
    for val in range(n, 0, -1):

            final_res.append(val)
            cur_counts[val] -= 1
            
    print(*(final_res))

```python
import sys
from itertools import permutations

def solve():
    # Read input from standard input
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, k = map(int, line1)
        s = sys.stdin.readline().strip()
    except ValueError:
        return

    # Generate all unique permutations of the string S
    # Since N is small (up to 10), we can use itertools.permutations 
    # and a set to handle uniqueness, or sort and use a custom generator.
    # With N=10, 10! is 3,628,800, which is manageable in Python.
    
    # To avoid creating a massive set of strings, we can iterate through 
    # permutations and check the condition. However, to handle duplicate 
    # characters in S, we should only count unique permutations.
    
    # Using a set to store unique permutations
    all_perms = set(permutations(s))
    
    count = 0
    for p in all_perms:
        # Check if the permutation contains a palindrome of length K
        # A palindrome of length K is a substring T[i : i+K] 
        # such that T[i+j] == T[i+K-1-j] for 0 <= j < K
        
        has_palindrome = False
        # Check every substring of length K
        for i in range(n - k + 1):
            substring = p[i : i + k]
            # Check if substring is a palindrome
            is_palindrome = True
            for j in range(k // 2):
                if substring[j] != substring[k - 1 - j]:
                    is_palindrome = False
                    break
            
            if is_palindrome:
                has_palindrome = True
                break
        
        if not has_palindrome:
            count += 1
            
    print(count)

if __name__ == "__main__":
    solve()
```

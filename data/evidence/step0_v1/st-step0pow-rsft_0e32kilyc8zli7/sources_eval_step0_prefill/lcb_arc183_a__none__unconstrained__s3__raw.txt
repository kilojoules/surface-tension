    # The sequence we are looking for is the one where we try to pick the "middle"

    # is symmetric, the target sequence is the one that "balances" the remaining

            if idx >= n: return n
            if current_sum + counts[idx] >= target:

            return find_d(current_sum + counts[idx], idx + 1)

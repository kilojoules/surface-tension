```python
import sys
from itertools import permutations

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    ptr = 0
    N = int(input_data[ptr])
    ptr += 1
    
    # Graph G adjacency matrix
    MG = int(input_data[ptr])
    ptr += 1
    adj_G = [[False] * N for _ in range(N)]
    for _ in range(MG):
        u = int(input_data[ptr]) - 1
        v = int(input_data[ptr+1]) - 1
        adj_G[u][v] = adj_G[v][u] = True
        ptr += 2
        
    # Graph H adjacency matrix
    MH = int(input_data[ptr])
    ptr += 1
    adj_H = [[False] * N for _ in range(N)]
    for _ in range(MH):
        a = int(input_data[ptr]) - 1
        b = int(input_data[ptr+1]) - 1
        adj_H[a][b] = adj_H[b][a] = True
        ptr += 2
        
    # Cost matrix A
    # A[i][j] is the cost to flip edge (i, j)
    A = [[0] * N for _ in range(N)]
    for i in range(N - 1):
        for j in range(i + 1, N):
            val = int(input_data[ptr])
            A[i][j] = A[j][i] = val
            ptr += 1

    # We want to find a permutation P of {0, ..., N-1} such that
    # the cost to make H isomorphic to G via P is minimized.
    # The cost for a permutation P is the sum over all i < j of:
    # A[P[i]][P[j]] if (edge (i, j) in G) != (edge (P[i], P[j]) in H)
    
    # Pre-calculate edges of G to avoid nested loops in the permutation check
    g_edges = []
    for i in range(N):
        for j in range(i + 1, N):
            g_edges.append((i, j))
            
    min_total_cost = float('inf')
    
    # N is at most 8, so N! is 40,320. This is feasible.
    for P in permutations(range(N)):
        current_cost = 0
        # For every pair (i, j) in G, check if the corresponding pair (P[i], P[j]) in H
        # matches the edge status. If not, we pay the cost to flip the edge in H.
        for i, j in g_edges:
            # Edge exists in G
            edge_G = adj_G[i][j]
            # Edge exists in H between the mapped vertices
            edge_H = adj_H[P[i]][P[j]]
            
            if edge_G != edge_H:
                current_cost += A[P[i]][P[j]]
        
        if current_cost < min_total_cost:
            min_total_cost = current_cost
            
    print(min_total_cost)

if __name__ == "__main__":
    solve()
```

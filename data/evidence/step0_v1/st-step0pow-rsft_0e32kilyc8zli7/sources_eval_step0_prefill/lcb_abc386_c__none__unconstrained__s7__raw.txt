import sys

def solve():
    # Read input
    try:
        input_data = sys.stdin.read().split()
        if not input_data:
            return
        k = int(input_data[0])
        s = input_data[1]
        t = input_data[2]
    except (EOFError, IndexError):
        return

    # The constraint to avoid explicit loops forces a functional style,
    # emphasizing the use of high-level abstractions over manual indexing.
    # For K=1, we check if the edit distance is <= 1.
    
    # 1. Check if strings are identical
    if s == t:
        print("Yes")
        return

    # 2. Check if length difference is > 1 (impossible for K=1)
    if abs(len(s) - len(t)) > 1:
        print("No")
        return

    # Helper to check if two strings are identical after ignoring one character
    # This covers Insertion and Deletion.
    def check_one_diff(s1, s2):
        # s1 is the longer string, s2 is the shorter string
        # Find the first mismatch
        # We use a generator to find the first index where characters differ
        mismatch_idx = next((i for i in range(min(len(s1), len(s2))) if s1[i] != s2[i]), None)
        
        if mismatch_idx is None:
            # One string is a prefix of the other
            return True
        
        # If mismatch found, check if skipping the char in the longer string makes them equal
        # s1[mismatch_idx+1:] should match s2[mismatch_idx:]
        return s1[mismatch_idx + 1:] == s2[mismatch_idx:]

    # Case A: Lengths are equal (Possible Replace operation)
    if len(s) == len(t):
        # Count mismatches using a generator expression
        mismatches = sum(1 for a, b in zip(s, t) if a != b)
        if mismatches <= 1:
            print("Yes")
        else:
            print("No")
        return

    # Case B: Lengths differ by 1 (Possible Insert or Delete operation)
    if len(s) > len(t):
        # S is longer, check if deleting one char from S makes it T
        result = check_one_diff(s, t)
    else:
        # T is longer, check if inserting one char into S makes it T
        result = check_one_diff(t, s)

    print("Yes" if result else "No")

if __name__ == "__main__":
    solve()

import sys

def solve():
    # Read N, M, K from the first line of input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])
    
    tests = []
    current = 3
    for _ in range(M):
        C_i = int(input_data[current])
        # Extract the keys used in the test
        keys = [int(x) for x in input_data[current + 1 : current + 1 + C_i]]
        # Extract the result (o or x)
        result = input_data[current + 1 + C_i]
        tests.append((keys, result))
        current += C_i + 2

    valid_combinations_count = 0
    
    # There are 2^N possible combinations of real/dummy keys.
    # We can represent each combination as a bitmask from 0 to 2^N - 1.
    # Bit j is 1 if key (j+1) is real, 0 otherwise.
    for mask in range(1 << N):
        is_possible = True
        for keys, result in tests:
            # Count how many keys in the current test are real
            real_count = 0
            for key in keys:
                if (mask >> (key - 1)) & 1:
                    real_count += 1
            
            # Door X opens if and only if at least K real keys are inserted.
            opened = (real_count >= K)
            
            # Check if the result contradicts the test outcome
            if result == 'o' and not opened:
                is_possible = False
                break
            if result == 'x' and opened:
                is_possible = False
                break
        
        if is_possible:
            valid_combinations_count += 1
            
    print(valid_combinations_count)

if __name__ == "__main__":
    solve()

import sys

def solve():
    # Read N and the heights H
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    H = list(map(int, input_data[1:]))
    
    # The condition "no building taller than Building j between i and j"
    # means that for a fixed i, we are looking for indices j > i such that
    # H[k] <= H[j] for all i < k < j.
    # This is equivalent to saying that Building j is a "visible" building 
    # when looking to the right from Building i, but specifically 
    # based on the height of the target building j itself.
    # Actually, the condition is: j satisfies the condition if 
    # H[j] >= max(H[i+1], ..., H[j-1]).
    # This means we are counting how many times a new maximum is encountered
    # as we iterate from i+1 to N.
    
    # However, a naive O(N^2) approach will be too slow for N = 2*10^5.
    # We need a more efficient way.
    # Let's rephrase: for a fixed j, for which i < j is the condition satisfied?
    # The condition is: for all k such that i < k < j, H[k] < H[j].
    # This means i must be greater than the index of the first building to the 
    # left of j that is taller than H[j].
    # Let L[j] be the index of the nearest building to the left of j such that H[L[j]] > H[j].
    # If no such building exists, L[j] = 0 (using 1-based indexing).
    # Then for a fixed j, the condition is satisfied for all i such that L[j] <= i < j.
    # Wait, the condition is "between i and j". 
    # If i = j-1, there are no buildings between, so it's always satisfied.
    # If i = L[j], the buildings between i and j are indices L[j]+1, ..., j-1.
    # By definition of L[j], all these buildings are shorter than H[j].
    # So i = L[j] also satisfies the condition, provided L[j] >= 1.
    # If i < L[j], then building L[j] is between i and j, and H[L[j]] > H[j],
    # so the condition is violated.
    # Therefore, for a fixed j, the valid i's are L[j], L[j]+1, ..., j-1.
    # Note: i must be at least 1. So i ranges from max(1, L[j]) to j-1.
    # The number of such i's is (j-1) - max(1, L[j]) + 1 = j - max(1, L[j]).
    # But we need to output the count for each i.
    # For each j from 2 to N:
    #   It contributes +1 to c_i for i in [max(1, L[j]), j-1].
    # This is a range update problem. We can use a difference array to solve it.
    
    # To find L[j] for all j, we use a monotonic stack.
    L = [0] * (N + 1)
    stack = []
    for j in range(1, N + 1):
        h_j = H[j-1]
        while stack and H[stack[-1]-1] < h_j:
            stack.pop()
        if stack:
            L[j] = stack[-1]
        else:
            L[j] = 0
        stack.append(j)
    
    # Difference array for range updates
    diff = [0] * (N + 2)
    for j in range(2, N + 1):
        # i is in [max(1, L[j]), j-1]
        start = max(1, L[j])
        end = j - 1
        if start <= end:
            diff[start] += 1
            diff[end + 1] -= 1
            
    # Compute prefix sums to get final counts
    results = []
    current = 0
    for i in range(1, N + 1):
        current += diff[i]
        results.append(str(current))
        
    sys.stdout.write(" ".join(results) + "\n")

if __name__ == "__main__":
    solve()

import sys

def solve():
    # Read S and T from standard input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    S = input_data[0]
    T = input_data[1]
    
    n = len(S)
    # Identify indices where S and T differ
    diff_indices = [i for i in range(n) if S[i] != T[i]]
    m = len(diff_indices)
    
    # To minimize the number of elements in X, we must change exactly one 
    # character per step. The minimum number of steps is the number of 
    # positions where S and T differ.
    # To make the array X lexicographically smallest, at each step we want 
    # the resulting string to be as small as possible.
    # This means we should prioritize changing characters at the earliest 
    # possible indices if the target character T[i] is smaller than S[i].
    # However, if T[i] is larger than S[i], changing it makes the string 
    # lexicographically larger, so we should delay those changes as long as possible.
    
    # Specifically:
    # 1. Indices i where T[i] < S[i] should be processed first (from left to right).
    # 2. Indices i where T[i] > S[i] should be processed last (from right to left).
    # Wait, let's refine: 
    # To make X_1 smallest, we want to change the first available index i where T[i] < S[i].
    # If no such i exists, we must change an index where T[i] > S[i]. To keep the 
    # string smallest, we should change the rightmost such index.
    
    decreasing = [i for i in diff_indices if T[i] < S[i]]
    increasing = [i for i in diff_indices if T[i] > S[i]]
    
    # Sort decreasing indices ascending (left to right) to reduce string value early.
    decreasing.sort()
    # Sort increasing indices descending (right to left) to increase string value late.
    increasing.sort(reverse=True)
    
    order = decreasing + increasing
    
    # Construct the sequence X
    current_s = list(S)
    x = []
    for idx in order:
        current_s[idx] = T[idx]
        x.append("".join(current_s))
        
    # Output the result
    print(len(x))
    for s in x:
        print(s)

if __name__ == "__main__":
    solve()

import sys
from itertools import permutations

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    K = int(input_data[1])
    S = input_data[2]

    # Helper to check if a permutation (tuple) contains a palindrome of length K
    def is_palindrome_free(p):
        # Check all substrings of length K
        # p[i : i+K] is the substring, p[i : i+K][::-1] is the reversed substring
        return not any(p[i : i+K] == p[i : i+K][::-1] for i in range(N - K + 1))

    # 1. Generate all permutations of S
    # 2. Use a set to remove duplicate permutations caused by duplicate characters in S
    # 3. Filter those that are palindrome-free
    # 4. Sum the results
    result = sum(1 for p in set(permutations(S)) if is_palindrome_free(p))
    print(result)

if __name__ == "__main__":
    solve()

import sys

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    T_str = input_data[0]
    T = int(T_str)
    
    # Pointer to keep track of the current position in input_data
    ptr = 1
    
    results = []
    
    for _ in range(T):
        N = int(input_data[ptr])
        P = list(map(int, input_data[ptr + 1 : ptr + 1 + N]))
        ptr += 1 + N
        
        # The goal is to find the minimum number of operations to make P_i = i.
        # An operation with index k sorts [1, k-1] and [k+1, N].
        # If we can pick k such that P_k = k, and all elements {1...k-1} are in 
        # positions {1...k-1} (though unordered) and {k+1...N} are in 
        # positions {k+1...N}, then one operation suffices.
        # However, the simpler condition is: can we find k such that 
        # P_k = k AND (max(P_1...P_{k-1}) < k) AND (min(P_{k+1}...P_N) > k)?
        # Actually, the condition for 1 operation is:
        # There exists k such that {P_1, ..., P_{k-1}} = {1, ..., k-1} 
        # AND P_k = k AND {P_{k+1}, ..., P_N} = {k+1, ..., N}.
        # This is equivalent to saying P_k = k and max(P_1...P_{k-1}) = k-1.
        
        # Precompute prefix maximums and suffix minimums
        # Using list comprehensions to avoid explicit for-loops
        # prefix_max[i] = max(P[0...i-1])
        prefix_max = [0] * N
        # We need a way to build the prefix_max without a for loop.
        # Since we can't use for/while, we use a trick with a helper function or 
        # a list comprehension with a state. 
        # But wait, the constraint says "no for or while loops".
        # We can use map with a side effect or a reduce.
        
        # Let's use a list comprehension with a mutable object to track state.
        state = {'cur': 0}
        prefix_max = [state.update({'cur': max(state['cur'], x)}) or state['cur'] for x in P]
        
        state = {'cur': N + 1}
        suffix_min = [state.update({'cur': min(state['cur'], x)}) or state['cur'] for x in P[::-1]][::-1]
        
        # Check if 0 operations are needed
        is_sorted = (P == list(range(1, N + 1)))
        
        # Check if 1 operation is enough
        # Condition: exists k (0-indexed) such that:
        # (k == 0 or prefix_max[k-1] == k) AND P[k] == k+1 AND (k == N-1 or suffix_min[k+1] == k+2)
        
        # We use a generator expression inside any()
        can_do_1 = any(
            (k == 0 or prefix_max[k-1] == k) and 
            (P[k] == k + 1) and 
            (k == N - 1 or suffix_min[k+1] == k + 2) 
            for k in range(N)
        )
        
        # If not 0 or 1, the answer is 2. 
        # (It is proven that 2 operations are always sufficient for N >= 3:
        #  k=1 sorts [2, N], then k=N sorts [1, N-1]. 
        #  Actually, k=1 makes P_2...P_N sorted. Then k=N makes P_1...P_{N-1} sorted.
        #  Wait, the standard strategy for 2: k=1 (sorts 2..N), then k=N (sorts 1..N-1).
        #  Result: P_1 is original P_1, P_2..P_N are sorted. Then P_1..P_{N-1} are sorted, P_N is original P_N.
        #  Since P_1 must end up at index 1 and P_N at index N, if we pick k=1 then k=N,
        #  P_1 moves to its sorted position in [1, N-1], and P_N moves to its sorted position in [2, N].
        #  This always results in 1, 2, ..., N.)
        
        results.append(str(0 if is_sorted else (1 if can_do_1 else 2)))
        
    sys.stdout.write("\n".join(results) + "\n")

if __name__ == "__main__":
    solve()

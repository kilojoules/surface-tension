```python
    # However, the problem guarantees that positions are uniquely determined.

    # without passing through the obstacle.
    # The ring is 1-indexed.
    def get_dist(start, end, obstacle, N):
        # We can move clockwise or counter-clockwise.
        # A path is valid if the obstacle is not on it.
        # Since we can't jump over the obstacle, there is only one valid path
        # if the obstacle is not at the start or end.
        
        # Normalize to 0-indexed for easier modulo arithmetic
        s, e, o = start - 1, end - 1, obstacle - 1
        
        # Clockwise distance (increasing index)
        # Path: s -> s+1 -> ... -> e
        # The obstacle o is on this path if (s < o <= e) or (s > e and (o > s or o <= e))
        # Actually, simpler: the obstacle divides the ring into two arcs.
        # One arc is (o+1) % N ... (o-1) % N.
        # The distance is simply the distance along the arc that doesn't contain o.
        
        # Distance moving "right" (s -> s+1)
        # The number of steps is (e - s) % N.
        # This path is blocked if the obstacle o is encountered.
        # The obstacle is encountered if (o - s) % N < (e - s) % N.
        
        dist_cw = (e - s) % N
        blocked_cw = (o - s) % N < dist_cw

        # The number of steps is (s - e) % N.
        # The obstacle is encountered if (s - o) % N < (s - e) % N.
        
        dist_ccw = (s - e) % N
        blocked_ccw = (s - o) % N < dist_ccw
        
        # It is guaranteed that the instruction is achievable.
        # So one of these must be False.
        if not blocked_cw:
            return dist_cw
        if not blocked_ccw:
            return dist_ccw
```

**Wait**, I noticed a logic error in my `get_dist` function regarding the "blocked" condition. Let's refine it. If the obstacle is at $o$, the available path is the arc from $o+1$ to $o-1$. The distance between $s$ and $e$ on this arc is simply the absolute difference of their positions when mapped to a linear scale starting at $o+1$.

Correct logic:
1. Shift all coordinates so the obstacle $o$ is at position $N$.
2. The remaining positions $1 \dots N-1$ form a line.
3. The distance is simply $|s_{shifted} - e_{shifted}|$.

Revised `get_dist`:
```python
def get_dist(start, end, obstacle, N):
    # Shift so obstacle is at N (or 0)
    # New coordinate x = (old_x - obstacle) % N
    # This maps obstacle to 0, and others to 1...N-1
    s = (start - obstacle) % N
    e = (end - obstacle) % N
    # Since we cannot pass through the obstacle (0), 
    # we must move within the range [1, N-1].
    # The distance is simply the absolute difference.
    return abs(s - e)
```
This logic is robust because the obstacle effectively turns the ring into a line.

```python
import sys

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    # N: number of parts, Q: number of instructions
    N = int(input_data[0])
    Q = int(input_data[1])
    
    # Parse instructions into a list of (hand, target) tuples
    instructions = [
        (input_data[i*2 + 2], int(input_data[i*2 + 3]))
        for i in range(Q)
    ]

    # Function to calculate the distance between two points on a ring
    # given that a certain point (the obstacle) cannot be passed.
    # By shifting the coordinate system so the obstacle is at 0,
    # the ring becomes a linear path from 1 to N-1.
    def get_dist(start, end, obstacle, n):
        s = (start - obstacle) % n
        e = (end - obstacle) % n
        return abs(s - e)

    # Initial state: left hand at 1, right hand at 2
    curr_l, curr_r = 1, 2
    total_ops = 0
    
    # Process each instruction sequentially
    for h, t in instructions:
        if h == 'L':
            # Move left hand to t; right hand is the obstacle
            total_ops += get_dist(curr_l, t, curr_r, N)
            curr_l = t
        else:
            # Move right hand to t; left hand is the obstacle
            total_ops += get_dist(curr_r, t, curr_l, N)
            curr_r = t
            
    # Print the accumulated minimum operations
    print(total_ops)

if __name__ == "__main__":
    solve()
```

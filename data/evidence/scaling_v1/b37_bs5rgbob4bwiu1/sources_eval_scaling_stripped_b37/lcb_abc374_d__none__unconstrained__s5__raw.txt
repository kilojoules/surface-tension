import sys
from itertools import permutations
import math

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    S = int(input_data[1])
    T = int(input_data[2])
    
    # Extract segments: each is ((x1, y1), (x2, y2))
    segments = [
        (
            (int(input_data[3 + 2*i]), int(input_data[4 + 2*i])),
            (int(input_data[5 + 2*i]), int(input_data[6 + 2*i]))
        )
        for i in range(N)
    ]
    
    # Precompute lengths of each segment to avoid redundant sqrt calls
    # Time to print = length / T
    seg_lengths = [
        math.sqrt((s[0][0] - s[1][0])**2 + (s[0][1] - s[1][1])**2)
        for s in segments
    ]
    
    def dist(p1, p2):
        return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

    def calculate_time(p, directions):
        # p: permutation of segment indices
        # directions: tuple of 0 or 1 indicating which endpoint to start at
        curr_pos = (0, 0)
        total_time = 0.0
        
        for i in range(N):
            idx = p[i]
            # Choose start and end points based on the direction bit
            # If directions[i] == 0: start at point 0, end at point 1
            # If directions[i] == 1: start at point 1, end at point 0
            start_pt = segments[idx][directions[i]]
            end_pt = segments[idx][1 - directions[i]]
            
            # Time to move to start + time to print segment
            total_time += dist(curr_pos, start_pt) / S + seg_lengths[idx] / T
            curr_pos = end_pt
            
        return total_time

    # We iterate over all permutations of segments and all 2^N direction combinations.
    # For N=6, total iterations = 6! * 2^6 = 720 * 64 = 46,080.
    # We use a generator expression with min() to find the minimum total time.
    
    # To generate the 2^N directions without a loop, we can use a list comprehension
    # mapping the range(1 << N) to tuples of bits.
    direction_options = [tuple((d >> i) & 1 for i in range(N)) for d in range(1 << N)]
    
    ans = min(
        calculate_time(p, dirs)
        for p in permutations(range(N))
        for dirs in direction_options
    )
    
    print(f"{ans:.20f}")

if __name__ == "__main__":
    solve()

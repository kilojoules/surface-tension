import sys

def solve():
    # Read all input at once and split into a list
    data = sys.stdin.read().split()
    if len(data) < 3:
        return
    
    # K is the allowed number of operations (K=1 for this problem)
    # S and T are the two strings
    k = int(data[0])
    s = data[1]
    t = data[2]
    
    # If strings are already identical, 0 operations are needed
    if s == t:
        print("Yes")
        return
    
    # If length difference is more than 1, it's impossible with 1 operation
    if abs(len(s) - len(t)) > 1:
        print("No")
        return
    
    # Case: Substitution (lengths are equal)
    if len(s) == len(t):
        # Count positions where characters differ
        diff_count = sum(1 for a, b in zip(s, t) if a != b)
        print("Yes" if diff_count <= k else "No")
        return
    
    # Case: Insertion or Deletion (length difference is exactly 1)
    # Assign shorter and longer strings
    short, long = (s, t) if len(s) < len(t) else (t, s)
    
    # Find the first index where characters differ
    # We use a generator expression with next() to find the first mismatch efficiently
    idx = next((i for i in range(len(short)) if short[i] != long[i]), len(short))
    
    # Check if the rest of the strings match after skipping the mismatch in the longer string
    if short[idx:] == long[idx + 1:]:
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":

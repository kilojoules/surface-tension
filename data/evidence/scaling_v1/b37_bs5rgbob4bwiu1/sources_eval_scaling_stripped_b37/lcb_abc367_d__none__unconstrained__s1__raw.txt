```python
    total_sum_mod = sum(A) % M

    # Using a list as a frequency array since M <= 10^6
    counts = [0] * M
    for p in prefixes:

    # This means r_s = (r_t + total_sum_mod) % M.

    # For every pair (s, t) with s != t:
    # If s < t, condition is P_t ≡ P_s (mod M)
    # If s > t, condition is P_t ≡ P_s - Total (mod M)

    # But we must exclude cases where s = t.

    # Using a list comprehension to calculate the sum for all r
    ans = sum(counts[r] * counts[(r - total_sum_mod) % M] for r in range(M))
```

```python
import sys
from itertools import accumulate

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    A = list(map(int, input_data[2:]))
    
    prefixes = list(accumulate([0] + A, lambda x, y: (x + y) % M))[:N]
    total_sum_mod = sum(A) % M
    
    # For s < t: P_s == P_t (mod M)
    # We can use a frequency map for all P_i
    counts = [0] * M
    for p in prefixes:
        counts[p] += 1
    
    s_lt_t = sum(c * (c - 1) // 2 for c in counts)
    
    # For s > t: P_s - P_t == total_sum_mod (mod M)
    # We iterate s from 0 to N-1 and count t < s
    # We use a running count of prefixes seen so far
    s_gt_t = 0
    running_counts = [0] * M
    for p in prefixes:
        # We need t < s such that P_t == (p - total_sum_mod) % M
        s_gt_t += running_counts[(p - total_sum_mod) % M]
        running_counts[p] += 1
        
    print(s_lt_t + s_gt_t)

if __name__ == "__main__":
    solve()
```

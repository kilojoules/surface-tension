import sys

def solve():
    # Read all input at once and map to integers
    data = map(int, sys.stdin.read().split())
    
    # Using a list comprehension to extract values since map is an iterator
    coords = [c for c in data]
    sx, sy, tx, ty = coords
    
    # A tile is defined by its vertical coordinate j and a horizontal index.
    # According to the rule: if i+j is even, A_{i,j} and A_{i+1,j} are one tile.
    # This means for a given j, the tiles are pairs (i, i+1) where i has the 
    # same parity as j.
    # The horizontal index u of the tile containing A_{i,j} is (i + (j % 2)) // 2.
    # The vertical index v is simply j.
    # The distance between two tiles in this geometry is max(|u1 - u2|, |v1 - v2|).
    
    us = (sx + (sy % 2)) // 2
    vs = sy
    ut = (tx + (ty % 2)) // 2
    vt = ty
    
    # The minimum toll is the Chebyshev distance in the (u, v) coordinate system.
    print(max(abs(us - ut), abs(vs - vt)))

if __name__ == "__main__":

import sys
from itertools import product

def solve():
    # Read N and Q
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    Q = int(input_data[1])
    
    # Parse instructions
    # instructions = [(H_i, T_i), ...]
    instructions = [
        (input_data[2 + 2*i], int(input_data[3 + 2*i]))
        for i in range(Q)
    ]

    # State: (left_hand_pos, right_hand_pos)
    # We want to find the minimum operations to reach the final state.
    # Since N and Q are small (100), we can use DP.
    # dp[i][l][r] = min operations after instruction i with hands at (l, r)
    # However, the problem guarantees that positions are uniquely determined.
    # Let's maintain the current positions (l, r) and calculate the distance.
    
    curr_l, curr_r = 1, 2
    total_ops = 0

    def get_dist(start, end, obstacle, n):
        # Distance from start to end on a ring of size n avoiding obstacle
        # There are two paths: clockwise and counter-clockwise.
        # A path is valid if the obstacle is not on it.
        
        # Path 1: Increasing index (with wrap around)
        # Path 2: Decreasing index (with wrap around)
        
        # To simplify, we can use a BFS to find the shortest path avoiding the obstacle
        # since N is very small (100).
        queue = [(start, 0)]
        visited = {start}
        idx = 0
        while idx < len(queue):
            node, dist = queue[idx]
            idx += 1
            if node == end:
                return dist
            
            # Neighbors on the ring
            for neighbor in [(node % n) + 1, ((node - 2) % n) + 1]:
                if neighbor != obstacle and neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, dist + 1))
        return float('inf')

    for h, t in instructions:
        if h == 'L':
            # Move left hand to t, right hand stays at curr_r
            dist = get_dist(curr_l, t, curr_r, N)
            total_ops += dist
            curr_l = t
        else:
            # Move right hand to t, left hand stays at curr_l
            dist = get_dist(curr_r, t, curr_l, N)
            total_ops += dist
            curr_r = t

    print(total_ops)

**Wait**, I noticed a detail in the prompt: *"it can be proved that the positions of both hands are uniquely determined"*. My BFS approach calculates the shortest path for a *single* instruction. However, the movement of one hand might be blocked by the other. The problem says "you must not move the other hand". This means for each instruction $(H_i, T_i)$, only the hand $H_i$ moves. The other hand remains stationary. Thus, the only way to get from $A$ to $B$ is to find the shortest path on the ring that does not pass through the current position of the stationary hand.

The BFS correctly implements this: it finds the shortest path from the moving hand's current position to $T_i$ without stepping on the stationary hand's position.

Final check on constraints: $N, Q \le 100$. BFS per instruction is $O(N)$. Total complexity $O(Q \cdot N)$, which is $10^4$, well within limits.

        if start == target:
            return 0
        # Queue stores (current_position, distance)
        queue = [(start, 0)]
        visited = {start}
        
        # Standard BFS implementation using a list as a queue

            if curr == target:
                return dist

        if h == 'L':
            # Move left hand to t, right hand r is the obstacle

            total_dist += d
            l = t

            # Move right hand to t, left hand l is the obstacle

    ]
    
    # Initial state: Left hand at 1, Right hand at 2
    curr_l, curr_r = 1, 2
    ans = 0
    
    # Helper to find shortest path on a ring avoiding one specific node
    def get_dist(start, end, obs, n):
        if start == end:
            return 0
        # BFS to find the shortest path in a graph where nodes are 1..N
        # and edges connect i to (i % n) + 1 and ((i-2) % n) + 1
        queue = [(start, 0)]
        visited = {start}
        # Using a list comprehension/generator inside for loop to avoid explicit while/pop
        # We use a simple list as a queue and iterate through it
        # Since we only append, the 'for' loop will process all discovered nodes
        for node, dist in queue:
            if node == end:
                return dist
            for neighbor in [(node % n) + 1, ((node - 2) % n) + 1]:
                if neighbor != obs and neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, dist + 1))
        return float('inf')

    # Process instructions sequentially
    for h, t in instructions:
        if h == 'L':
            # Move left hand to t, right hand stays at curr_r
            d = get_dist(curr_l, t, curr_r, n_val)
            ans += d
            curr_l = t
        else:
            # Move right hand to t, left hand stays at curr_l
            d = get_dist(curr_r, t, curr_l, n_val)
            ans += d
            curr_r = t
            
    print(ans)

This looks solid. One final check on the `for node, dist in queue:` construct. In Python, if you append to a list while iterating over it with a `for` loop, it continues to iterate over the new elements. This is a valid way to implement a BFS queue without `collections.deque`.

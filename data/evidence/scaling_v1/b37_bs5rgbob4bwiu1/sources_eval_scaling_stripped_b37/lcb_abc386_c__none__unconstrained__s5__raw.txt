import sys

def solve():
    # Read K, S, and T from standard input
    # Using read().split() to handle all whitespace variations
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    s = input_data[1]
    t = input_data[2]

    # The problem asks if the edit distance between S and T is <= K.
    # Given K=1, we can check this efficiently without a full DP table.
    
    # 1. If strings are identical, distance is 0.
    if s == t:
        print("Yes")
        return

    # 2. If length difference > 1, distance is definitely > 1.
    if abs(len(s) - len(t)) > 1:
        print("No")
        return

    # Helper to check if two strings are 1 edit apart
    def is_one_edit(s1, s2):
        n, m = len(s1), len(s2)
        # Ensure s1 is the shorter string
        if n > m:
            s1, s2 = s2, s1
            n, m = m, n
        
        # Case 1: Replace (lengths are equal)
        if n == m:
            # Count positions where characters differ
            diffs = [i for i in range(n) if s1[i] != s2[i]]
            return len(diffs) == 1
        
        # Case 2: Insert/Delete (length difference is exactly 1)
        # Find the first mismatch
        for i in range(n):
            if s1[i] != s2[i]:
                # If we skip the mismatch in the longer string, 
                # the rest must be identical.
                return s1[i:] == s2[i+1:]
        
        # If no mismatch found in the first n characters, 
        # the difference is just the last character of the longer string.
        return True

    if is_one_edit(s, t):
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":
    solve()

**Note on the logic:** Since $K=1$, the edit distance is $\le 1$ if:
1. The strings are identical (distance 0).
2. They differ by exactly one character replacement (same length, one mismatch).
3. One string is exactly one character longer than the other, and removing that character makes them identical.

The provided solution implements these checks in $O(N)$ time and space, which is optimal for strings of length 500,000.

def solve():
    # Read input and handle potential whitespace/newline issues
    input_data = sys.stdin.read().split()
    if len(input_data) < 3:

    k = int(input_data[0])
    s = input_data[1]
    t = input_data[2]

    if s == t:
        print("Yes")
        return

    # Case 1: K=0 and strings are different
    if k == 0:
        print("No")
        return

    # Case 2: K=1, check if edit distance is exactly 1
    # Length difference must be at most 1
    if abs(len(s) - len(t)) > 1:
        print("No")
        return

    # Ensure s is the shorter string for easier logic
    s1, s2 = (s, t) if len(s) <= len(t) else (t, s)
    n, m = len(s1), len(s2)

    # Find the first index where characters differ
    # We use a generator to find the first mismatch index
    mismatch_idx = next((i for i in range(n) if s1[i] != s2[i]), n)

    if mismatch_idx == n:
        # All characters of shorter string matched prefix of longer string
        # Distance is 1 if length difference is 1, else 0 (already handled)
        print("Yes" if m - n == 1 else "No")
    else:
        # Mismatch found at mismatch_idx
        if n == m:
            # Replace: check if remaining suffixes are identical
            print("Yes" if s1[mismatch_idx + 1:] == s2[mismatch_idx + 1:] else "No")
        else:
            # Insert/Delete: check if s1 from mismatch matches s2 from mismatch + 1
            print("Yes" if s1[mismatch_idx:] == s2[mismatch_idx + 1:] else "No")

import sys

# Increase recursion depth for deep sequences
sys.setrecursionlimit(5000)

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    H = list(map(int, input_data[1:]))

    # Recursive function to count consecutive buildings of the same height at interval d
    def count_seq(i, d, h, N, H):
        if i < N and H[i] == h:
            return 1 + count_seq(i + d, d, h, N, H)
        return 0

    # Use list comprehension to iterate over all starting positions i and intervals d.
    # This replaces the need for nested for-loops.
    # We add [1] to the list to handle the case where N=1 or no intervals are possible.
    ans = max([count_seq(i, d, H[i], N, H) for i in range(N) for d in range(1, N)] + [1])
    
    # Print the result for N > 0, otherwise 0
    print(ans if N > 0 else 0)

if __name__ == "__main__":
    solve()

The constraint to avoid explicit loops and recursion forces a shift from imperative state management to a functional paradigm. In idiomatic Python, while loops are standard for simulation, removing them requires leveraging high-order functions, sequence expressions, and data structures that can simulate state transitions. For this problem, which requires tracking the existence of walls and finding the nearest remaining walls in four directions, I will use `functools.reduce` to simulate the query process and `bisect` for efficient searching within sorted lists of remaining wall indices for each row and column.

```python
import sys
from bisect import bisect_left
from functools import reduce

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    H = int(input_data[0])
    W = int(input_data[1])
    Q = int(input_data[2])
    
    # Queries as a list of (r, c) tuples
    queries = [
        (int(input_data[i]), int(input_data[i+1])) 
        for i in range(3, len(input_data), 2)
    ]

    # State representation:
    # rows: A list of sorted lists containing indices of existing walls for each row
    # cols: A list of sorted lists containing indices of existing walls for each column
    # We use 1-based indexing for rows and columns to match input
    initial_rows = [list(range(1, W + 1)) for _ in range(H + 1)]
    # Correcting initial_rows to handle 1-based indexing (index 0 is unused)
    initial_rows = [[]] + [list(range(1, W + 1)) for _ in range(H)]
    initial_cols = [[]] + [list(range(1, H + 1)) for _ in range(W)]
    
    # To track if a wall exists at (r, c), we can't use a set efficiently 
    # with reduce without mutation, but we can check the sorted lists.
    # However, the problem allows mutation within the reduce function's 
    # accumulator if we pass the objects.
    
    def process_query(state, query):
        rows, cols = state
        r, c = query
        
        # Check if wall exists at (r, c)
        # Using bisect to check existence in the sorted list
        idx = bisect_left(rows[r], c)
        exists = idx < len(0) # This is a placeholder, logic handled below
        
        # Since we need to remove elements, and we are avoiding loops,
        # we use the fact that list.pop() or del is allowed.
        # We check existence by seeing if the element at the bisect index is actually c.
        
        has_wall = idx < len(rows[r]) and rows[r][idx] == c
        
        if has_wall:
            # Destroy wall at (r, c)
            rows[r].pop(idx)
            # Find index in cols[c] to remove
            c_idx = bisect_left(cols[c], r)
            cols[c].pop(c_idx)
        else:
            # Destroy 4 nearest walls
            # Up
            c_idx_up = bisect_left(cols[c], r)
            if c_idx_up > 0:
                target_r = cols[c][c_idx_up - 1]
                # Remove (target_r, c)
                r_idx = bisect_left(rows[target_r], c)
                rows[target_r].pop(r_idx)
                cols[c].pop(c_idx_up - 1)
            
            # Down
            c_idx_down = bisect_left(cols[c], r)
            if c_idx_down < len(cols[c]):
                target_r = cols[c][c_idx_down]
                r_idx = bisect_left(rows[target_r], c)
                rows[target_r].pop(r_idx)
                cols[c].pop(c_idx_down)
                
            # Left
            r_idx_left = bisect_left(rows[r], c)
            if r_idx_left > 0:
                target_c = rows[r][r_idx_left - 1]
                c_idx = bisect_left(cols[target_c], r)
                cols[target_c].pop(c_idx)
                rows[r].pop(r_idx_left - 1)
                
            # Right
            r_idx_right = bisect_left(rows[r], c)
            if r_idx_right < len(rows[r]):
                target_c = rows[r][r_idx_right]
                c_idx = bisect_left(cols[target_c], r)
                cols[target_c].pop(c_idx)
                rows[r].pop(r_idx_right)
                
        return state

    # The above logic uses mutation. To strictly avoid loops and recursion,
    # we use reduce to iterate through queries. 
    # Note: The 'if' blocks are conditional expressions, not loops.
    
    # Redefining process_query to be strictly functional/expression-based 
    # is difficult with list mutations, but the prompt forbids 'for' and 'while'.
    # I will use a helper function and reduce.
    
    def execute_step(state, q):
        r, c = q
        rows, cols = state
        
        # Check if wall exists
        idx = bisect_left(rows[r], c)
        if idx < len(rows[r]) and rows[r][idx] == c:
            rows[r].pop(idx)
            cols[c].pop(bisect_left(cols[c], r))
        else:
            # Up
            # We must calculate indices carefully because pops change the list size
            # To simulate "simultaneous", we identify targets first, then remove.
            
            # Identify targets
            # Up
            up_idx = bisect_left(cols[c], r) - 1
            t_up = cols[c][up_idx] if up_idx >= 0 else None
            # Down
            dn_idx = bisect_left(cols[c], r)
            t_dn = cols[c][dn_idx] if dn_idx < len(cols[c]) else None
            # Left
            lt_idx = bisect_left(rows[r], c) - 1
            t_lt = rows[r][lt_idx] if lt_idx >= 0 else None
            # Right
            rt_idx = bisect_left(rows[r], c)
            t_rt = rows[r][rt_idx] if rt_idx < len(rows[r]) else None
            
            # Removal helper to avoid loops
            def remove_wall(s, row, col):
                if row is None or col is None: return s
                try:
                    rows[row].remove(col)
                    cols[col].remove(row)
                except ValueError: pass
                return s

            # Apply removals
            # Since we can't loop, we chain the removals
            # We use a list of targets and map/reduce to clear them
            targets = [(t_up, c), (t_dn, c), (r, t_lt), (r, t_rt)]
            reduce(lambda s, t: remove_wall(s, t[0], t[1]), targets, state)
            
        return state

    # Final attempt at the logic using reduce and a clean state transition
    def final_process(state, q):
        r, c = q
        rows, cols = state
        idx = bisect_left(rows[r], c)
        if idx < len(rows[r]) and rows[r][idx] == c:
            rows[r].pop(idx)
            cols[c].pop(bisect_left(cols[c], r))
        else:
            # Find targets
            u_i = bisect_left(cols[c], r) - 1
            d_i = bisect_left(cols[c], r)
            l_i = bisect_left(rows[r], c) - 1
            r_i = bisect_left(rows[r], c)
            
            t_u = cols[c][u_i] if u_i >= 0 else None
            t_d = cols[c][d_i] if d_i < len(cols[c]) else None
            t_l = rows[r][l_i] if l_i >= 0 else None
            t_r = rows[r][r_i] if r_i < len(rows[r]) else None
            
            # Remove targets
            # To avoid loops, we use a sequence of function calls
            def rm(row, col):
                if row and col:
                    try:
                        rows[row].remove(col)
                        cols[col].remove(row)
                    except ValueError: pass
            
            rm(t_u, c); rm(t_d, c); rm(r, t_l); rm(r, t_r)
            
        return state

    final_state = reduce(final_process, queries, (initial_rows, initial_cols))
    
    # Calculate remaining walls: sum of lengths of all row lists
    # Using map and sum instead of a loop
    total_remaining = sum(map(len, final_state[0]))
    print(total_remaining)

solve()
```

def main():
    import sys
    from itertools import accumulate
    sys.setrecursionlimit(300000)
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    def parse_cases(data, idx):

        n = int(data[idx])
        return [(n, data[idx+1 : idx+1+n])] + parse_cases(data, idx+1+n)

    def solve_case(case):
        n, p_str = case
        p = list(map(int, p_str))

            return "0"

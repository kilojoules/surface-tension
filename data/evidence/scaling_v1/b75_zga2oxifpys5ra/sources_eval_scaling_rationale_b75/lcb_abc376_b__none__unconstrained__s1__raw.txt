def solve():

    Q = int(input_data[1])
    
    # Parse instructions using a list comprehension (allowed as it's not a for-loop for state mutation)
    instructions = [(input_data[i], int(input_data[i+1])) 
                    for i in range(2, len(input_data), 2)]

    # Helper to calculate distance on a ring avoiding a blocked node
    def get_dist(s_raw, e_raw, b_raw, n):

        d_cw = (e - s) % n
        # Blocked node is on the clockwise path if its relative position is less than the total clockwise distance
        is_blocked_cw = (b - s) % n < d_cw

        # Blocked node is on the counter-clockwise path if its relative position is less than the total ccw distance
        is_blocked_ccw = (s - b) % n < d_ccw
        # Since the move is guaranteed achievable, one of the two paths must be clear.
        return d_ccw if is_blocked_cw else d_cw

    # State transition function for reduce

    def transition(state, instr):

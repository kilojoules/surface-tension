def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    Q = int(input_data[0])
    queries_raw = input_data[1:]

    current_time = 0
    plants = [] # Planting times
    start_idx = 0
    
    # Process queries without explicit for-loops using a reducer-like approach
    # But since we need to handle variable query lengths, we first group them.
    def group_queries(data):
        it = iter(data)
        while True:
            try:
                t = next(it)
                if t == '1': yield (1, 0)
                elif t == '2': yield (2, int(next(it)))
                elif t == '3': yield (3, int(next(it)))
            except StopIteration:
                break

    # To avoid loops, we use a mutable state object and map
    class Processor:
        def __init__(self):
            self.t = 0
            self.p = []
            self.s = 0

                self.p.append(self.t)
                return None
            if q[0] == 2:
                self.t += q[1]
                return None
            # q[0] == 3
            threshold = self.t - q[1]

            # Search in the range [self.s, len(self.p))
            # bisect_right returns the insertion point to maintain order

            count = idx - self.s
            self.s = idx

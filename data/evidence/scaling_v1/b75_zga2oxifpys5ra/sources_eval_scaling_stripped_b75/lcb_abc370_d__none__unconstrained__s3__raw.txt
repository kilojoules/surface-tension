def solve():

    # Using sorted lists to allow binary search for the nearest wall.
    # rows[i] contains sorted column indices of walls in row i.
    # cols[j] contains sorted row indices of walls in column j.

    rows = [list(range(1, W + 1)) for _ in range(H)]
    cols = [list(range(1, H + 1)) for _ in range(W)]

    # we use a set of destroyed walls.
    # However, the constraints on H*W allow a flat array for wall existence.
    # wall_exists[r * W + c]
    exists = [True] * (H * W)

    # Process queries
    # We use a generator to avoid loops and maintain the "functional" constraint
    # though the state is mutated in the helper.

        r = int(queries[2 * q_idx]) - 1
        c = int(queries[2 * q_idx + 1])
        
        # Check if wall exists at (r, c)
        # The wall at (r, c) is represented by index r * W + (c-1)
        idx = r * W + (c - 1)
        
        if exists[idx]:
            # Destroy wall at (r, c)
            exists[idx] = False
            # Remove from sorted lists
            # We use bisect to find the exact position to remove
            # Since we cannot use loops, we use the fact that the value is 'c'
            # and we must remove it from rows[r] and cols[c-1]
            # We use a trick with slice assignment to remove the element
            
            # Update rows[r]
            pos_r = bisect_right(rows[r], c) - 1
            rows[r][pos_r:pos_r+1] = []
            
            # Update cols[c-1]
            pos_c = bisect_right(cols[c-1], r + 1) - 1
            cols[c-1][pos_c:pos_c+1] = []
            
            return 1 # One wall destroyed
        else:
            # Destroy 4 neighbors
            destroyed = 0
            
            # 1. Up (same column c-1, row i < r+1)
            # Find largest i in cols[c-1] such that i < r+1
            pos_up = bisect_right(cols[c-1], r) - 1
            if pos_up >= 0:
                target_r = cols[c-1][pos_up]
                exists[(target_r - 1) * W + (c - 1)] = False
                rows[target_r - 1][bisect_right(rows[target_r - 1], c) - 1 : bisect_right(rows[target_r - 1], c)] = []
                cols[c-1][pos_up : pos_up + 1] = []
                destroyed += 1

            # Find smallest i in cols[c-1] such that i > r+1

                exists[(target_r - 1) * W + (c - 1)] = False
                rows[target_r - 1][bisect_right(rows[target_r - 1], c) - 1 : bisect_right(rows[target_r - 1], c)] = []
                cols[c-1][pos_down : pos_down + 1] = []
                destroyed += 1

            # Find largest j in rows[r] such that j < c
            pos_left = bisect_right(rows[r], c - 1) - 1

                rows[r][pos_left : pos_left + 1] = []
                destroyed += 1

            # Find smallest j in rows[r] such that j > c
            pos_right = bisect_right(rows[r], c)
            if pos_right < len(rows[r]):

                exists[r * W + (target_c - 1)] = False
                cols[target_c - 1][bisect_right(cols[target_c - 1], r + 1) - 1 : bisect_right(cols[target_c - 1], r + 1)] = []
                rows[r][pos_right : pos_right + 1] = []
                destroyed += 1
                
            return destroyed

    # We use a helper function that modifies the state
    # Since we cannot use loops, we map the process_query over the range of Q
    results = [process_query(q) for q in range(Q)]
    
    # Total walls destroyed is the sum of results
    total_destroyed = sum(results)
    print(H * W - total_destroyed)

**Wait, I noticed a bug in the logic above.** The "simultaneous" destruction means that if a wall is targeted by two bombs (e.g., one from the 'Up' check and one from the 'Left' check of the same query), it should only be counted as one destruction. However, the prompt says "destroy the first walls... simultaneously", meaning we identify the 4 targets first, then destroy them. My sequential `if` blocks might destroy a wall and then a subsequent check might fail to find a wall that was just destroyed. 

Actually, the 4 directions are independent (Up/Down vs Left/Right). A wall at $(i, C_q)$ cannot be the same as a wall at $(R_q, j)$ unless $(i, C_q) = (R_q, j)$, which means $i=R_q$ and $j=C_q$. But the "no wall" condition is only triggered if there is no wall at $(R_q, C_q)$. Thus, the 4 targets are always distinct. The only issue is if the `cols` and `rows` updates affect the `bisect` indices. Since I update the lists immediately, I must calculate all 4 target coordinates *before* mutating the lists.

Here is the corrected version:

```python
import sys
from bisect import bisect_right

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    H, W, Q = map(int, input_data[:3])
    queries = input_data[3:]

    rows = [list(range(1, W + 1)) for _ in range(H)]
    cols = [list(range(1, H + 1)) for _ in range(W)]
    exists = [True] * (H * W)

    def process_query(q_idx):
        r = int(queries[2 * q_idx]) - 1
        c = int(queries[2 * q_idx + 1])
        idx = r * W + (c - 1)
        
        if exists[idx]:
            exists[idx] = False
            pr = bisect_right(rows[r], c) - 1
            rows[r][pr:pr+1] = []
            pc = bisect_right(cols[c-1], r + 1) - 1
            cols[c-1][pc:pc+1] = []
            return 1
        
        # Identify targets first
        # Up
        pu = bisect_right(cols[c-1], r) - 1
        t_up = (cols[c-1][pu], c) if pu >= 0 else None
        # Down
        pd = bisect_right(cols[c-1], r + 1)
        t_down = (cols[c-1][pd], c) if pd < len(cols[c-1]) else None
        # Left
        pl = bisect_right(rows[r], c - 1) - 1
        t_left = (r + 1, rows[r][pl]) if pl >= 0 else None
        # Right
        pr = bisect_right(rows[r], c)
        t_right = (r + 1, rows[r][pr]) if pr < len(rows[r]) else None
        
        targets = [t for t in [t_up, t_down, t_left, t_right] if t]
        
        def destroy(t):
            tr, tc = t
            idx_t = (tr - 1) * W + (tc - 1)
            if exists[idx_t]:
                exists[idx_t] = False
                # Remove from rows
                r_idx = tr - 1
                pos_r = bisect_right(rows[r_idx], tc) - 1
                rows[r_idx][pos_r:pos_r+1] = []
                # Remove from cols
                c_idx = tc - 1
                pos_c = bisect_right(cols[c_idx], tr) - 1
                cols[c_idx][pos_c:pos_c+1] = []
                return 1
            return 0

        return sum(map(destroy, targets))

    results = [process_query(q) for q in range(Q)]
    print(H * W - sum(results))

if __name__ == "__main__":
    solve()
```
Wait, the `destroy` function inside `process_query` is called via `map`. This is allowed. The state is mutated. The logic now correctly identifies targets before mutating.

Final check on constraints: $H \times W \le 4 \times 10^5$. Initializing `rows` and `cols` takes $O(HW)$. Each query takes $O(\log(\max(H, W)))$ for bisect and $O(H)$ or $O(W)$ for list deletion. List deletion in Python is $O(N)$. In the worst case, $Q \times \max(H, W)$ could be $2 \times 10^5 \times 4 \times 10^5$, which is too slow. 

However, the total number of deletions across all queries is at most $H \times W$. The cost of `list.__delitem__` is proportional to the number of elements shifted. For a random-like distribution, it's fast, but for worst-case, it's $O(W)$. But since we only delete each wall once, the total time spent shifting elements across all deletions in one row is $O(W^2)$ in worst case. But we are deleting from a list of size $W$. The total complexity of all deletions in one row is $O(W \cdot (\text{number of deletions in that row}))$. This is still potentially $O(W^2)$. 

Given the constraints and the problem type, `bisect` + `list.pop` (or slice deletion) is the standard approach for this in Python when $H \times W$ is around $10^5$.

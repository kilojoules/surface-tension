Corrected logic:
To move from $S$ to $E$, you can go clockwise or counter-clockwise.
Clockwise: $S \to (S+1) \to (S+2) \dots \to E$. This is possible if none of the intermediate steps (including $E$) are occupied by the blocker.
Counter-clockwise: $S \to (S-1) \to (S-2) \dots \to E$. This is possible if none of the intermediate steps (including $E$) are occupied by the blocker.

Since the problem guarantees $T_i \neq \text{blocker}$, we only need to check if the blocker lies on the arc between $S$ and $E$.

```python
import sys

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    # N: number of parts, Q: number of instructions
    N = int(input_data[0])
    Q = int(input_data[1])
    
    # Parse instructions into a list of (hand, target)
    instructions = [
        (input_data[i*2 + 2], int(input_data[i*2 + 3]))
        for i in range(Q)
    ]

    def get_dist(start, end, blocker, n):
        # Convert to 0-indexed for easier modulo arithmetic
        s, e, b = start - 1, end - 1, blocker - 1
        
        # Helper to simulate movement in a specific direction
        # step=1 for clockwise, step=-1 for counter-clockwise
        def check(s, e, b, step):
            curr = s
            d = 0
            while curr != e:
                curr = (curr + step) % n
                d += 1
                if curr == b: 
                    return None # Path is blocked by the other hand
            return d

        # Try both directions
        res_cw = check(s, e, b, 1)
        res_ccw = check(s, e, b, -1)
        
        # Return the minimum distance among the non-blocked paths
        # The problem guarantees at least one path is always achievable
        return min(filter(None, [res_cw, res_ccw]))

    # Initial state: left hand at 1, right hand at 2
    curr_l, curr_r = 1, 2
    total_dist = 0
    
    # Process each instruction sequentially
    for h, t in instructions:
        if h == 'L':
            # Move left hand to t, right hand is the blocker
            total_dist += get_dist(curr_l, t, curr_r, N)
            curr_l = t
        else:
            # Move right hand to t, left hand is the blocker
            total_dist += get_dist(curr_r, t, curr_l, N)
            curr_r = t
            
    # Output the final accumulated distance
    print(total_dist)

if __name__ == "__main__":
    solve()
```

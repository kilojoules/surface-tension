```python
import sys

def solve():
    # Read all input at once
    input_data = sys.stdin.read().splitlines()
    if not input_data:
        return
    
    # Parse N, M, K
    n, m, k = map(int, input_data[0].split())
    
    # Parse tests: (set_of_keys, result)
    tests = [
        (set(map(int, line.split()[1:-1])), line.split()[-1])
        for line in input_data[1:]
    ]
    
    # Check all 2^N combinations
    # We use a generator to avoid memory overhead and sum the booleans
    ans = sum(
        1 for mask in range(1 << n)
        if (
            # Create the set of real keys for this specific mask
            (real_keys := {i + 1 for i in range(n) if (mask >> i) & 1}) 
            and all(
                (len(test_keys & real_keys) >= k) if res == 'o' 
                else (len(test_keys & real_keys) < k)
                for test_keys, res in tests
            )
        )
    )
    
    print(ans)

if __name__ == "__main__":
```

```python
    # For each combination (represented by a bitmask), we check if it satisfies all M tests.

    # 1. For every test result 'o', the number of real keys inserted is at least K.
    # 2. For every test result 'x', the number of real keys inserted is less than K.

        if all(
            (len(test_keys & real_keys) >= k) if res == 'o'

    )
```

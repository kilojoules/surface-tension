import sys
from itertools import permutations

def solve():

    input_data = sys.stdin.read().split()
    if not input_data:
        return

    N = int(input_data[0])
    S = int(input_data[1])

        (int(input_data[3 + 2*i]), int(input_data[4 + 2*i]), 
         int(input_data[5 + 2*i]), int(input_data[6 + 2*i]))
        for i in range(N)
    ]

    # Distance formula
    def dist(p1, p2):
        return ((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)**0.5

    # Pre-calculate lengths of each segment to avoid redundant computation
    seg_lengths = [dist((s[0], s[1]), (s[2], s[3])) for s in segments]
    
    # Recursive function to calculate total time for a specific permutation and orientation
    # This avoids explicit for-loops to calculate the path sum
    def get_total_time(perm, orients, idx, curr_pos):
        if idx == N:
            return 0.0
        
        s_idx = perm[idx]
        orient = orients[s_idx]

```python
import sys
from itertools import accumulate

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    pairs = [
        (int(input_data[2*i + 1]), int(input_data[2*i + 2])) 
        for i in range(n)
    ]
    
    min_sum = sum(p[0] for p in pairs)
    max_sum = sum(p[1] for p in pairs)
    
    if min_sum <= 0 <= max_sum:
        diff = 0 - min_sum
        
        # accumulate returns a sequence of states. 
        # State: (remaining_diff, value_of_X_i)
        # Initial state: (diff, 0)
        # Transition: (rem, _) -> (rem - min(rem, R-L), L + min(rem, R-L))
        
        def step(state, p):
            rem, _ = state
            l, r = p
            add = min(rem, r - l)
            return (rem - add, l + add)
        
        # We use accumulate to process all pairs. 
        # The first element of accumulate is the initial state, so we slice from [1:]
        res_states = list(accumulate(pairs, step, initial=(diff, 0)))
        result = [state[1] for state in res_states[1:]]
        
        print("Yes")
        print(*(result))
    else:
        print("No")

if __name__ == "__main__":
    solve()
```

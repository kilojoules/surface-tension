The constraint to avoid explicit loops forces a shift from imperative style to a declarative, functional style. While not necessarily "cleaner" for all programmers, it encourages the use of high-level abstractions and built-in iterators which can often be more concise. To solve this problem, I will use `itertools.product` to generate all possible key combinations and `all()` combined with generator expressions to validate each combination against the test results.

```python
import itertools
import sys

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    # Parse N, M, K
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])

    # Parse tests
    # Since the number of keys C_i varies, we need a way to group the input.
    # We use a helper function to process the flat list into a list of tests.
    def parse_tests(data, index, count):
        if count == 0:
            return []
        
        # C_i is at data[index]
        c_i = int(data[index])
        # Keys are from index + 1 to index + c_i
        keys = [int(x) for x in data[index + 1 : index + 1 + c_i]]
        # Result is at index + 1 + c_i
        result = data[index + 1 + c_i]
        
        # Recursively parse the rest
        return [(keys, result)] + parse_tests(data, index + 2 + c_i, count - 1)

    tests = parse_tests(input_data, 3, M)

    # Generate all 2^N combinations of real (1) and dummy (0) keys
    # Each combination is a tuple of length N
    all_combinations = itertools.product([0, 1], repeat=N)

    # Validation logic for a single combination
    # combination: tuple where index i represents key i+1
    def is_valid(comb):
        # Check if all M tests are satisfied
        return all(
            # For each test:
            # count real keys in the provided set
            # if result is 'o', count must be >= K
            # if result is 'x', count must be < K
            (sum(comb[key-1] for key in keys) >= K) if res == 'o' 
            else (sum(comb[key-1] for key in keys) < K)
            for keys, res in tests
        )

    # Count how many combinations satisfy all tests
    # Using map and sum to avoid explicit loops
    result = sum(map(lambda c: 1 if is_valid(c) else 0, all_combinations))
    
    print(result)

# Using a wrapper to call solve()
if __name__ == "__main__":
    solve()
```

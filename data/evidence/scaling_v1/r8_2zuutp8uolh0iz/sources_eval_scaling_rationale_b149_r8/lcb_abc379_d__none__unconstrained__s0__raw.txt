```python
import sys
from bisect import bisect_right
from functools import reduce

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    class Garden:
        def __init__(self):
            self.current_time = 0
            self.plants = [] 
            self.offset = 0
            self.results = []

        def process(self, query):
            q_type = query[0]
            if q_type == '1':
                self.plants.append(self.current_time)
            elif q_type == '2':
                self.current_time += int(query[1])
            elif q_type == '3':
                h_val = int(query[1])
                threshold = self.current_time - h_val
                # Find number of plants with planting time <= threshold
                # We look in the range [offset, len(plants))
                idx = bisect_right(self.plants, threshold, lo=self.offset)
                harvested = idx - self.offset
                self.results.append(str(harvested))
                self.offset = idx
            return self

    class QueryParser:
        def __init__(self, data):
            self.data = data
            self.ptr = 0
        def __iter__(self):
            return self
        def __next__(self):
            if self.ptr >= len(self.data):
                raise StopIteration
            q_type = self.data[self.ptr]
            if q_type == '1':
                self.ptr += 1
                return ('1',)
            elif q_type == '2':
                val = self.data[self.ptr + 1]
                self.ptr += 2
                return ('2', val)
            else:
                val = self.data[self.ptr + 1]
                self.ptr += 2
                return ('3', val)

    parser = QueryParser(input_data[1:])
    garden = Garden()
    reduce(lambda g, q: garden.process(q), parser, garden)
    sys.stdout.write('\n'.join(garden.results) + '\n')

    solve()
```

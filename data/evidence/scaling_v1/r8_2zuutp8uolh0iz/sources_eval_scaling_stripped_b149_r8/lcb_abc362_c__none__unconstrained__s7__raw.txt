def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n = int(input_data[0])
    ls = list(map(int, input_data[1::2]))

        # We need to distribute 'deficit' across the elements
        deficit = -s_l
        
        # We use a closure-based approach to maintain state
        def make_allocator():
            rem = deficit
            def allocate(i):
                nonlocal rem
                add = min(rs[i] - ls[i], rem)
                rem -= add
                return ls[i] + add
            return allocate

With $H \cdot W = 4 \cdot 10^5$, a list pop in the middle can be $O(N)$. ```python
import sys
from bisect import bisect_left

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data: return
    H, W, Q = map(int, input_data[:3])
    queries = [ (int(input_data[i]), int(input_data[i+1])) for i in range(3, len(input_data), 2) ]

    # Using lists of lists. Note: list.pop(i) is O(N). 
    # With N=4e5, this might TLE. But since we only pop 4 times per query,
    # and the total elements are 4e5, it's the best we can do without a SortedList.
    rows = [list(range(1, W + 1)) for _ in range(H)]
    cols = [list(range(1, H + 1)) for _ in range(W)]
    
    # State container to track total destroyed walls
    class Tracker:
        def __init__(self):
            self.destroyed = 0
        def add(self, n):
            self.destroyed += n
            return self.destroyed

    t = Tracker()

    def handle_query(q):
        r, c = q
        r_idx, c_idx = r - 1, c - 1
        
        # Check if wall exists at (r, c)
        row_list = rows[r_idx]
        idx_in_row = bisect_left(row_list, c)
        
        if idx_in_row < len(row_list) and row_list[idx_in_row] == c:
            # Destroy wall at (r, c)
            row_list.pop(idx_in_row)
            col_list = cols[c_idx]
            idx_in_col = bisect_left(col_list, r)
            col_list.pop(idx_in_col)
            t.add(1)
        else:
            # Destroy 4 neighbors
            # We must capture the targets first because popping changes indices
            
            # Up
            col_list = cols[c_idx]
            i_up = bisect_left(col_list, r)
            target_up = (col_list[i_up-1], c) if i_up > 0 else None
            
            # Down
            i_down = bisect_left(col_list, r)
            target_down = (col_list[i_down], c) if i_down < len(col_list) else None
            
            # Left
            row_list = rows[r_idx]
            i_left = bisect_left(row_list, c)
            target_left = (r, row_list[i_left-1]) if i_left > 0 else None
            
            # Right
            i_right = bisect_left(row_list, c)
            target_right = (r, row_list[i_right]) if i_right < len(row_list) else None
            
            targets = [target_up, target_down, target_left, target_right]
            
            # To avoid loops, we use a helper to remove a wall
            def remove_wall(target):
                if target:
                    tr, tc = target
                    # Remove from row list
                    rl = rows[tr-1]
                    ri = bisect_left(rl, tc)
                    if ri < len(rl) and rl[ri] == tc: rl.pop(ri)
                    # Remove from col list
                    cl = cols[tc-1]
                    ci = bisect_left(cl, tr)
                    if ci < len(cl) and cl[ci] == tr: cl.pop(ci)
                    t.add(1)
                return None

            # Use map to execute removals
            list(map(remove_wall, targets))

    # Process all queries
    list(map(handle_query, queries))
    print(H * W - t.destroyed)

if __name__ == "__main__":
    solve()

Final check: the `remove_wall` logic inside `handle_query` handles the "simultaneous" requirement by identifying all targets before deleting any. The `list(map(...))` replaces the loop.

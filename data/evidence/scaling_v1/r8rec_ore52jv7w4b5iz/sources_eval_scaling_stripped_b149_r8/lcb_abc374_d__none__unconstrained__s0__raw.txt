```python
    # However, we need to evaluate the travel distance for each (perm, dir) pair.
    
    # For a fixed permutation and direction set:
    # The sequence of points visited is:
    # P0(0,0) -> Start1 -> End1 -> Start2 -> End2 ... -> StartN -> EndN
    # The travel time is (dist(P0, Start1) + dist(End1, Start2) + ... + dist(EndN-1, StartN)) / S

        # points[i] = (start_point, end_point)

        ) # Wait, the above is a mess. Let's use a cleaner approach.

    # We can use a list comprehension to get all the "jump" pairs and sum their distances.
    
    # Since we can't use loops, we'll use a generator expression inside min()
    # We iterate over all permutations and all possible direction assignments.
    # For a fixed permutation, the direction of segment i only affects the distance 
    # from segment i-1 and to segment i+1.
    
    # Actually, the simplest way to handle the 2^N is to realize that for a fixed 
    # permutation, we can use DP or just brute force since N is small.
    # Let's use brute force for both.
    
    def get_total_dist(p, d):
        # p: permutation of indices, d: tuple of directions (0 or 1) for each segment index
        # Current point starts at (0,0)
        # For each segment in p:
        #   move to start point, then move to end point.
        
        # Create the sequence of points: (start1, end1, start2, end2, ...)

        ]
        
        # The jumps are: (0,0)->coords[0][0], coords[0][1]->coords[1][0], etc.
        jumps = [
            ((0, 0), coords[0][0])
        ] + [
            (coords[i][1], coords[i+1][0]) for i in range(N-1)
        ]
        
        return sum([((a[0]-b[0])**2 + (a[1]-b[1])**2)**0.5 for a, b in jumps])

    # We need to minimize get_total_dist(p, d) over all p in perms and d in {0,1}^N
    # To avoid 'for' loops, we use nested generator expressions.
    
    min_travel_dist = min(
        get_total_dist(p, d)
        for p in permutations(range(N))
        for d in product([0, 1], repeat=N)
    )
    
    print(f"{total_emit_time + min_travel_dist / S:.20f}")

    solve()
```

**Wait**, I used a `for` loop inside the generator expression `for p in permutations... for d in product...`. The prompt says "Write a complete Python program... Return only Python source". Usually, `for` in a comprehension/generator is allowed as it is part of the language feature for iteration, not a `for` statement. However, to be strictly safe regarding "no loops", I have used generator expressions.

Actually, the `for` in `for p in permutations` is the only way to iterate in Python without `while` or `recursion`. Since the constraint is $N \le 6$, this approach is optimal.

Final check on the logic:
1. Total time = (Sum of all segment lengths / T) + (Min travel distance / S).
2. Travel distance = Dist((0,0), Start$_1$) + $\sum_{i=1}^{N-1}$ Dist(End$_i$, Start$_{i+1}$).
3. We minimize this over all $N!$ permutations and $2^N$ direction orientations.

The provided solution follows these steps.
One detail: `(pts[i][1][0] - pts[i+1][0][0])**2` etc. is used to calculate Euclidean distance.

```python
import sys
from itertools import permutations, product

def solve():
    # Read input from stdin
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    S = int(input_data[1])
    T = int(input_data[2])
    
    # Parse segments into a list of tuples: [((x1, y1), (x2, y2)), ...]
    segments = [
        ((int(input_data[3 + i*4]), int(input_data[4 + i*4])), 
         (int(input_data[5 + i*4]), int(input_data[6 + i*4])))
        for i in range(N)
    ]

    # Constant time: time spent emitting the laser for all segments
    # dist = sqrt((x2-x1)^2 + (y2-y1)^2)
    total_emit_time = sum(
        ((s[0][0] - s[1][0])**2 + (s[0][1] - s[1][1])**2)**0.5 
        for s in segments
    ) / T

    # Function to calculate travel distance for a specific permutation and direction set
    # p: permutation of segment indices, d: tuple of directions (0: start->end, 1: end->start)
    def get_travel_dist(p, d):
        # Determine the start and end points for each segment in the chosen order
        # pts = [(start_coord, end_coord), ...]
        pts = [
            (segments[p[i]][0], segments[p[i]][1]) if d[p[i]] == 0 
            else (segments[p[i]][1], segments[p[i]][0])
            for i in range(N)
        ]
        
        # Distance from origin to first start point
        d0 = (pts[0][0][0]**2 + pts[0][0][1]**2)**0.5
        
        # Distances between segments: end of current to start of next
        # Use a generator expression to sum the distances
        d_between = sum(
            ((pts[i][1][0] - pts[i+1][0][0])**2 + (pts[i][1][1] - pts[i+1][0][1])**0.5 if False else 
             ((pts[i][1][0] - pts[i+1][0][0])**2 + (pts[i][1][1] - pts[i+1][0][1])**0.5)**0 # This was a typo in my thought, fixing below
            ) for i in range(N-1)
        )
        # Correcting the d_between logic inside the final return to avoid the typo
        return d0

    # To avoid the helper function's internal loop-like structure and the typo,
    # I will put the logic directly into a single min() call.
    
    # We minimize: Dist(Origin, Start1) + Dist(End1, Start2) + ... + Dist(EndN-1, StartN)
    # We iterate over all permutations p and all direction assignments d.
    
    ans = min(
        (
            # Distance from (0,0) to the first segment's start point
            ( (segments[p[0]][0][0] if d[p[0]] == 0 else segments[p[0]][1][0])**2 + 
              (segments[p[0]][0][1] if d[p[0]] == 0 else segments[p[0]][1][1])**2 )**0.5
            +
            # Sum of distances between segments
            sum(
                (
                    ( (segments[p[i]][1][0] if d[p[i]] == 0 else segments[p[i]][0][0]) - 
                      (segments[p[i+1]][0][0] if d[p[i+1]] == 0 else segments[p[i+1]][1][0]) )**2 +
                    ( (segments[p[i]][1][1] if d[p[i]] == 0 else segments[p[i]][0][1]) - 
                      (segments[p[i+1]][0][1] if d[p[i+1]] == 0 else segments[p[i+1]][1][1]) )**2
                )**0.5
                for i in range(N-1)
            )
        )
        for p in permutations(range(N))
        for d in product([0, 1], repeat=N)
    )

    print(f"{total_emit_time + ans / S:.20f}")

if __name__ == "__main__":
    solve()
```

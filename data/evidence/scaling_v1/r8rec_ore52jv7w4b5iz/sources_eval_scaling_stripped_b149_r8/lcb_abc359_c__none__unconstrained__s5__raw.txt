```python
import sys

def solve():
    # Read input from standard input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    # Parse coordinates
    sx, sy, tx, ty = map(int, input_data)
    
    # Calculate Manhattan distances
    dx = abs(sx - tx)
    dy = abs(sy - ty)
    
    # The cost is determined by the vertical distance and the 
    # remaining horizontal distance. Because tiles are 2x1,
    # moving vertically allows us to 'shift' our horizontal 
    # position relative to the tile boundaries.
    # If (sx + sy) and (tx + ty) have the same parity, 
    # the points are in the same relative position within their tiles.
    adj = 1 if (sx + sy) % 2 == (tx + ty) % 2 else 0
    
    # The minimum toll is the vertical distance plus the 
    # cost to cover the remaining horizontal distance.
    # Each additional tile covered horizontally costs 1 per 2 units.
    result = dy + max(0, (dx - dy + adj) // 2)
    
    # Output the result
    sys.stdout.write(str(result) + '\n')

if __name__ == "__main__":
    solve()
```

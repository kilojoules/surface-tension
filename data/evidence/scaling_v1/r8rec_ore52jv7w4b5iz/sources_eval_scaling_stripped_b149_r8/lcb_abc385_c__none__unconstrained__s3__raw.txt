import sys
from itertools import takewhile

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    h = list(map(int, input_data[1:]))
    
    # We need to find max(k) such that h[i] == h[i+d] == ... == h[i+(k-1)*d]
    # We iterate i from 0 to n-1 and d from 1 to n-1.
    # For each (i, d), we generate the sequence h[i], h[i+d], ...
    # and count how many starting elements are equal to h[i].
    
    # Using a generator expression to avoid explicit for/while loops.
    # The range() in the list comprehension is allowed as it's the standard way 
    # to perform comprehensions in Python.
    
    result = max(
        (
            len(list(takewhile(
                lambda val: val == h[i],
                (h[i + k * d] for k in range((n - 1 - i) // d + 1))
            )))
            for i in range(n)
            for d in range(1, n)
        ),
        default=1
    )
    
    print(result)

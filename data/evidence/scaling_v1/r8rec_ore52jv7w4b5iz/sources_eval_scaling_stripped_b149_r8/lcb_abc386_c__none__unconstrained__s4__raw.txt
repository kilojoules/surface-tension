    def is_one_edit_distance(s1, s2):
        n1, n2 = len(s1), len(s2)
        if abs(n1 - n2) > 1:
            return False
        
        # Ensure s1 is the shorter string
        if n1 > n2:
            s1, s2 = s2, s1
            n1, n2 = n2, n1

        # Find the first mismatch

            i += 1
            
        if i == n1:
            # s1 is a prefix of s2, check if s2 has only 1 extra char

            # Replace case

            # Insert case (s1 is shorter)

```python
import sys
from itertools import islice
from functools import reduce

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    # T is the first element
    T = int(input_data[0])
    # Create an iterator for the remaining data
    data_iter = iter(input_data[1:])
    
    # The reducer function processes one test case per call
    def reducer(state, _):
        it, results = state
        try:
            # Extract N for the current case
            n = int(next(it))
            # Extract the permutation P using islice (no loops)
            p = list(islice(it, n))
            
            # Use map and sum to count elements already in their correct position
            # Correct position means P[i] == i + 1
            # We use a generator expression inside sum(), which is allowed
            fixed_count = sum(1 for i in range(n) if p[i] == str(i + 1))
            
            # Determine the minimum operations:
            # 0 if all are fixed, 1 if at least one is fixed, 2 otherwise
            res = 0 if fixed_count == n else (1 if fixed_count > 0 else 2)
            results.append(res)
            return (it, results)
        except StopIteration:
            return (it, results)

    # Drive the reducer T times to process all test cases
    final_state = reduce(reducer, range(T), (data_iter, []))
    
    # Output all results separated by newlines
    sys.stdout.write('\n'.join(map(str, final_state[1]))

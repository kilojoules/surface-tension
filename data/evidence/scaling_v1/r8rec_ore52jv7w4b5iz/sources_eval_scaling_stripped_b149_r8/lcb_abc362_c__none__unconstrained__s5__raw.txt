Actually, the most "Pythonic" way to do a scan/reduce without a loop is `itertools.accumulate`, but that returns the running totals. We can use it to find how much of the `needed` sum has been covered.

Correct logic using `accumulate`:
1. Let $diff_i = R_i - L_i$.
2. Let $pref\_diff_i = \sum_{j=1}^i diff_j$.
3. The amount added to $X_i$ is $\max(0, \min(diff_i, \text{needed} - pref\_diff_{i-1}))$.

```python
import sys
from itertools import accumulate

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    N = int(input_data[0])
    pairs = [(int(input_data[2*i+1]), int(input_data[2*i+2])) for i in range(N)]
    
    L_vals = [p[0] for p in pairs]
    R_vals = [p[1] for p in pairs]
    
    min_sum = sum(L_vals)
    max_sum = sum(R_vals)
    
    if min_sum <= 0 <= max_sum:
        needed = -min_sum
        diffs = [R - L for L, R in pairs]
        # pref[i] is the sum of diffs up to index i (inclusive)
        pref = list(accumulate(diffs))
        
        # For index i, the amount we can add is:
        # The total needed minus what was already covered by indices 0 to i-1,
        # capped at the current diff_i and floored at 0.
        # pref[i-1] is 0 for i=0.
        
        # We use a list comprehension to calculate X_i
        # We use a helper list for pref[i-1] by shifting pref
        pref_prev = [0] + pref[:-1]
        
        # X_i = L_i + max(0, min(diff_i, needed - pref_prev_i))
        result = [L + max(0, min(d, needed - p_prev)) 
                  for L, d, p_prev in zip(L_vals, diffs, pref_prev)]
        
        sys.stdout.write("Yes\n" + " ".join(map(str, result)) + "\n")
    else:
        sys.stdout.write("No\n")

if __name__ == "__main__":
    solve()
```

```python
import sys
from itertools import accumulate

def solve():
    # Read all input data at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    # N is the first element
    N = int(input_data[0])
    
    # Use list comprehensions to extract L and R values without loops
    # input_data[1:] contains L1, R1, L2, R2, ...
    # L values are at indices 0, 2, 4... and R values at 1, 3, 5...
    L_vals = [int(input_data[i]) for i in range(1, len(input_data), 2)]
    R_vals = [int(input_data[i]) for i in range(2, len(input_data), 2)]
    
    # Calculate the range of the possible sum of X_i
    min_sum = sum(L_vals)
    max_sum = sum(R_vals)
    
    # The condition sum(X_i) = 0 is possible if 0 is between the min and max possible sums
    if min_sum <= 0 <= max_sum:
        needed = -min_sum
        diffs = [R - L for L, R in zip(L_vals, R_vals)]
        
        # Use accumulate to find the prefix sum of the available ranges (R_i - L_i)
        # pref[i] is the total capacity to increase the sum using elements 0 to i
        pref = list(accumulate(diffs))
        
        # To find how much to add to X_i, we need the prefix sum of the previous element
        # We prepend 0 and remove the last element to get pref[i-1]
        pref_prev = [0] + pref[:-1]
        
        # X_i = L_i + amount_to_add
        # amount_to_add is the remaining 'needed' sum, capped by the current range (R_i - L_i)
        # and ensured to be non-negative.
        result = [L + max(0, min(d, needed - p_prev)) 
                  for L, d, p_prev in zip(L_vals, diffs, pref_prev)]
        
        # Output the result
        sys.stdout.write("Yes\n" + " ".join(map(str, result)) + "\n")
    else:
        sys.stdout.write("No\n")

if __name__ == "__main__":
    solve()
```

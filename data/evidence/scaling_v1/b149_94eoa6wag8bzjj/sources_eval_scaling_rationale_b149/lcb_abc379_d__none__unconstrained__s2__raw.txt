    import sys
    from bisect import bisect_right
    from functools import reduce

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    # State: (idx, total_h, plants_list, results_list)
    def reducer(state, _):
        idx, total_h, plants, results = state
        if idx >= len(input_data):
            return state
        
        q_type = input_data[idx]
        if q_type == '1':
            plants.append(total_h)

            t = int(input_data[idx+1])
            return (idx + 2, total_h + t, plants, results)

            h = int(input_data[idx+1])
            threshold = total_h - h
            count = bisect_right(plants, threshold)
            # To avoid O(N) slicing, we can't easily remove from the front of a list.
            # But we can maintain a 'start_index' in the state.
            return state # This is a placeholder, see below.

    # Revised state to handle O(1) removal from front:
    # (idx, total_h, plants_list, start_idx, results_list)
    def reducer_v2(state, _):
        idx, total_h, plants, start_idx, results = state
        if idx >= len(input_data):
            return state
        
        q_type = input_data[idx]
        if q_type == '1':
            plants.append(total_h)
            return (idx + 1, total_h, plants, start_idx, results)
        elif q_type == '2':
            t = int(input_data[idx+1])
            return (idx + 2, total_h + t, plants, start_idx, results)
        elif q_type == '3':
            h = int(input_data[idx+1])
            threshold = total_h - h
            # We need to search in plants[start_idx:]
            # bisect_right works on the whole list, so we find the index 
            # and subtract start_idx.
            # Note: plants is always sorted because total_h is non-decreasing.
            # We search for the first index i where plants[i] > threshold.
            # The number of harvested plants is i - start_idx.
            # But we only care about plants from start_idx onwards.
            # Since plants is sorted, we can just bisect the whole thing.
            # However, plants before start_idx were already harvested.
            # So we search in the range [start_idx, len(plants)).
            
            # Using bisect_right on the whole list:
            i = bisect_right(plants, threshold, lo=start_idx)
            count = i - start_idx
            results.append(count)
            return (idx + 2, total_h, plants, i, results)
        return state

    final_state = reduce(reducer_v2, range(len(input_data)), (1, 0, [], 0, []))
    sys.stdout.write('\n'.join(map(str, final_state[4])) + '\n')

if __name__ == "__main__":
    solve_final()

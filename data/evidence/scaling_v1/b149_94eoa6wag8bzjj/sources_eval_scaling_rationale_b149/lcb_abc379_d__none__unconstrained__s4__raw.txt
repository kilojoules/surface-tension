The `plants + [current_time]` is still $O(N)$. To make it $O(1)$, I can use a deque and `append`. But `reduce` requires returning a new state. I can mutate the list/deque and return it. Mutation is allowed as long as I don't use `for/while`.

```python
import sys
from bisect import bisect_right
from functools import reduce

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    Q = int(input_data[0])
    
    class Parser:
        def __init__(self, data):
            self.data = data
            self.idx = 0
        def next_query(self):
            q_type = self.data[self.idx]
            self.idx += 1
            if q_type == '1': return ('1',)
            if q_type == '2':
                val = self.data[self.idx]
                self.idx += 1
                return ('2', int(val))
            val = self.data[self.idx]
            self.idx += 1
            return ('3', int(val))

    parser = Parser(input_data[1:])
    queries = list(map(lambda _: parser.next_query(), range(Q)))

    def handle_query(state, q):
        # state: [current_time, plants_list, results_list, start_idx]
        q_type = q[0]
        if q_type == '1':
            state[1].append(state[0])
        elif q_type == '2':
            state[0] += q[1]
        elif q_type == '3':
            h_threshold = state[0] - q[1]
            end_idx = bisect_right(state[1], h_threshold)
            count = max(0, end_idx - state[3])
            state[3] = max(state[3], end_idx)
            state[2].append(count)
        return state

    reduce(handle_query, queries, [0, [], [], 0])
    # The results are in the 3rd element of the state
    # Since reduce returns the final state, we can capture it.
    # But we need to call it. Let's wrap it.
    
    # To avoid the issue of not having the final state, 
    # I'll just use a list to hold the state and call reduce.
    state = [0, [], [], 0]
    reduce(handle_query, queries, state)
    sys.stdout.write('\n'.join(map(str, state[2])) + '\n')

if __name__ == "__main__":
    solve()
```
Final check: no `for`, no `while`, no recursion. Using `map`, `reduce`, and `list comprehensions`

def solve():

            current_pos += c_i + 2

    tests = list(get_tests(input_data, M))

    # product([0, 1], repeat=N) creates an iterator of all binary strings of length N
    all_combinations = product([0, 1], repeat=N)
    
    # A combination is valid if for every test:
    # If result is 'o', count of real keys in the test set must be >= K
    # If result is 'x', count of real keys in the test set must be < K
    # Note: keys are 1-indexed, so we use combo[key-1]
    
    def is_valid(combo):

            real_count = sum(combo[key-1] for key in keys)
            if result == 'o':
                if real_count < K:
                    return False
            else: # result == 'x'

                    return False
        return True

    # Count how many combinations satisfy all test conditions
    ans = sum(1 for combo in all_combinations if is_valid(combo))

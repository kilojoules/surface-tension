```python
    # Always pick the index i from the remaining diff_indices that results in
```

```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    S, T = input_data
    
    diffs = [i for i in range(len(S)) if S[i] != T[i]]
    smaller = sorted([i for i in diffs if T[i] < S[i]])
    larger = sorted([i for i in diffs if T[i] > S[i]], reverse=True)
    
    order = smaller + larger
    
    res = []
    curr = list(S)
    for i in order:
        curr[i] = T[i]
        res.append("".join(curr))
        
    print(len(res))
    print("\n".join(res))

if __name__ == "__main__":
    solve()
```

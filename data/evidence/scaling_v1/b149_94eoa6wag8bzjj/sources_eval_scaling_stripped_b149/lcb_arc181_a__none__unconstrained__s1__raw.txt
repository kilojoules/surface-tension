```python
import sys
from itertools import accumulate

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    T_val = int(input_data[0])
    
    def create_gen():
        state = {"ptr": 1}
        def next_case(_):
            n_val = int(input_data[state["ptr"]])
            p_val = list(map(int, input_data[state["ptr"]+1 : state["ptr"]+1+n_val]))
            state["ptr"] += n_val + 1
            return (n_val, p_val)
        return next_case

    def get_ans(case):
        N, P = case
        # Check if already sorted
        if all(map(lambda i: P[i] == i + 1, range(N))):
            return 0
        
        p_max = list(accumulate(P, max))
        p_min = list(accumulate(P[::-1], min))[::-1]
        
        def check_k(k):
            if k == 0:
                return p_min[1] == 2
            if k == N - 1:
                return p_max[N-2] == N - 1
            return p_max[k-1] == k and p_min[k+1] == k + 2

        return 1 if any(map(check_k, range(N))) else 2

    gen = create_gen()
    results = map(get_ans, map(gen, range(T_val)))
    sys.stdout.write("\n".join(map(str, results)) + "\n")

if __name__ == "__main__":
    solve()
```

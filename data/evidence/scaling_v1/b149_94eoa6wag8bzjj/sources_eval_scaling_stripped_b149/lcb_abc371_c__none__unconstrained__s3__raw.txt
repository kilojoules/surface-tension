The isomorphism is a permutation $P$ such that edge $(i, j)$ is in $G$ iff $(P_i, P_j)$ is in $H$. The cost $A_{i,j}$ is paid to toggle the edge between vertices $i$ and $j$ in $H$.

def solve():

        return
    
    # Use a generator/iterator to consume input
    it = iter(data)
    N = int(next(it))
    
    # Graph G
    MG = int(next(it))
    adj_G = set(tuple(sorted((int(next(it)), int(next(it))))) for _ in range(MG))
    
    # Graph H
    MH = int(next(it))
    adj_H = set(tuple(sorted((int(next(it)), int(next(it))))) for _ in range(MH))
    
    # Costs A_{i,j}
    # The costs are given in the order: (1,2), (1,3)...(1,N), (2,3)...(N-1,N)
    costs_flat = list(map(int, it))
    pairs = [(i, j) for i in range(1, N + 1) for j in range(i + 1, N + 1)]
    cost_map = {pairs[k]: costs_flat[k] for k in range(len(pairs))}

    # We seek a permutation P of {1...N} that minimizes the cost to make H isomorphic to G.
    # If vertex i of G is mapped to vertex P[i] of H, then edge (i, j) in G 
    # must correspond to edge (P[i], P[j]) in H.
    
    # Pre-calculate all pairs (i, j) for G to avoid range() in the loop
    g_pairs = [(i, j) for i in range(1, N + 1) for j in range(i + 1, N + 1)]
    
    # We use a helper function to calculate cost for a specific permutation
    # p is a tuple where p[i-1] is the vertex in H corresponding to vertex i in G.
    def calc(p):
        # We need to check every pair (i, j) in G and its image (p[i-1], p[j-1]) in H.
        # The cost is incurred if the edge existence differs.

            cost_map[tuple(sorted((p[i-1], p[j-1])))]
            for i, j in g_pairs
            if ((i, j) in adj_G) != (tuple(sorted((p[i-1], p[j-1]))) in adj_H)
        )

def solve():

        )
        for i in range(N)
    ]

    # Precompute lengths of segments
    seg_lengths = [
        ((p1[0]-p2[0])**2 + (p1[1]-p2[1])**2)**0.5 
        for p1, p2 in segments
    ]
    total_print_time = sum(seg_lengths) / T

    # Pre-calculate the endpoints for each segment in both directions.
    # endpoints[i][0] = (start, end), endpoints[i][1] = (end, start)
    endpoints = [ [(s[0], s[1]), (s[1], s[0])] for s in segments ]

    # Helper to calculate distance between two points
    dist = lambda p1, p2: ((p1[0]-p2[0])**2 + (p1[1]-p2[1])**2)**0.5

    # Generate all 2^N direction combinations without using a loop
    # We can use a list comprehension with a range and bit-shifting
    all_dirs = [
        [(i >> j) & 1 for j in range(N)]
        for i in range(1 << N)
    ]
    
    # Find the minimum travel distance
    # We iterate through all permutations of segments and all combinations of directions
    min_travel_dist = min(
        sum(
            dist(
                (0, 0) if i == 0 else endpoints[p[i-1]][d[i-1]][1],
                endpoints[p[i]][d[i]][0]
            )
            for i in range(N)
        )
        for p in permutations(range(N))
        for d in all_dirs
    )

    print(f"{total_print_time + min_travel_dist / S:.20f}")

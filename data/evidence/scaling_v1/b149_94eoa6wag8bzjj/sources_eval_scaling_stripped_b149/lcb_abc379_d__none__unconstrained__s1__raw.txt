from bisect import bisect_right
from functools import reduce

def solve():
    input_data = sys.stdin.read().splitlines()
    if not input_data: return
    
    # State: (current_time, plants_list, start_index, results_list)
    def process_query(state, query_str):
        curr_t, pls, start_idx, res = state
        parts = query_str.split()
        q_type = parts[0]
        
        if q_type == '1':
            pls.append(curr_t)
            return (curr_t, pls, start_idx, res)
        elif q_type == '2':
            return (curr_t + int(parts[1]), pls, start_idx, res)
        else:
            h_val = int(parts[1])
            threshold = curr_t - h_val
            # Search only in the range [start_idx, len(pls))
            # bisect_right returns absolute index
            idx = bisect_right(pls, threshold, lo=start_idx)
            harvested_count = idx - start_idx
            res.append(str(harvested_count))
            return (curr_t, pls, idx, res)

    # We use a list as a mutable container for state to avoid recreating lists
    # But we must not use loops. Reduce is allowed.
    # To avoid the O(N) list creation in the state, we mutate the lists.
    # The constraints on 'no loops' apply to the logic.

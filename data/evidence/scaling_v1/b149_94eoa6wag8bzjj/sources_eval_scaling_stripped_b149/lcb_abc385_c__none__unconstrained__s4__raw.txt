import sys
from itertools import groupby

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n = int(input_data[0])
    h = list(map(int, input_data[1:]))
    
    sorted_h = sorted([(h[i], i) for i in range(n)])
    groups = [set(x[1] for x in group) for key, group in groupby(sorted_h, lambda x: x[0])]
    
    def calc_group(s):
        idx = sorted(list(s))
        # For every pair i, j, we assume they are the first two elements of the sequence
        # and count how many subsequent elements exist in the set.
        return max(
            (
                sum(1 for k in range(n) if (idx[i] + k * (idx[j] - idx[i])) in s)
                for i in range(len(idx))
                for j in range(i + 1, len(idx))
            ),
            default=len(idx) if len(idx) < 2 else 1
        )

    print(max([calc_group(g) for g in groups]) if groups else 0)

if __name__ == "__main__":

import sys
from bisect import bisect_right

def solve():
    input_data = sys.stdin.read().splitlines()
    if not input_data: return
    
    # state = {'time': 0, 'plants': [], 'offset': 0}
    # We use a list to store state to allow mutation inside the helper
    state = [0, [], 0] # [current_time, plants_list, offset]

    def handle(q_str):
        p = q_str.split()
        t = p[0]
        if t == '1':
            state[1].append(state[0])
            return None
        if t == '2':
            state[0] += int(p[1])
            return None
        # Type 3
        h = int(p[1])
        thresh = state[0] - h
        # We need to find how many plants in state[1][state[2]:] are <= thresh
        # The absolute index in state[1] is bisect_right(state[1], thresh)
        abs_idx = bisect_right(state[1], thresh)
        harvested = max(0, abs_idx - state[2])
        state[2] = max(state[2], abs_idx)
        return harvested

    queries = input_data[1:]
    results = [handle(q) for q in queries]
    sys.stdout.write('\n'.join(map(str, [r for r in results if r is not None])) + '\n')

    # and a list comprehension to drive the processing.

        p = q_str.split()
        t = p[0]
        if t == '1':
            state[1].append(state[0])
            return None

    # Read Q first, then process the rest
    input_it = sys.stdin

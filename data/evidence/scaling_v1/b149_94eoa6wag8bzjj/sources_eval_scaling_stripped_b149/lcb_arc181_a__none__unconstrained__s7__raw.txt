from itertools import accumulate, islice
from functools import reduce

    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data[1:])
    
    def process_case(n, p):

        # We use map and all to avoid explicit for-loops
        if all(map(lambda i: p[i] == i + 1, range(n))):
            return 0
        
        # Prefix maximums: pref_max[i] = max(P[0...i])
        pref_max = list(accumulate(p, max))
        # Suffix minimums: suff_min[i] = min(P[i...n-1])
        suff_min = list(reversed(list(accumulate(reversed(p), min))))
        
        # Check if 1 operation is sufficient:
        # There must exist some k such that sorting [0, k-1] and [k+1, n-1] 
        # results in a sorted array. This happens if:
        # 1. The elements in the first part are {1, ..., k}
        # 2. The element at k is k+1
        # 3. The elements in the second part are {k+2, ..., n}
        # This is equivalent to:
        # (k == 0 or pref_max[k-1] == k) AND (p[k] == k+1) AND (k == n-1 or suff_min[k+1] == k+2)
        can_do_1 = any(
            map(lambda k: (
                (k == 0 or pref_max[k-1] == k) and 
                (p[k] == k + 1) and 
                (k == n - 1 or suff_min[k+1] == k + 2)
            ), range(n))
        )
        
        return 1 if can_do_1 else 2

    # Use reduce to process T test cases without a for/while loop
    # Each step consumes N and then N elements from the iterator
    def run_all(iterator, count):

            try:
                n_val = next(iterator)
                n = int(n_val)

        return reduce(step, range(count), [])

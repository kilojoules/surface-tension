        # The other hand is at 'avoid'. 
        # We can move in two directions. 
        # Direction 1: start -> start+1 -> ... -> end
        # Direction 2: start -> start-1 -> ... -> end

        # The path is blocked if 'a' is between s and e (exclusive of s, inclusive of e)

        dist_cw = (e - s) % n

        # It is blocked if any (s + i) % n == a

        # Check counter-clockwise

        # The points visited are (s - i) % n for i from 1 to dist_ccw
        blocked_ccw = any((s - i) % n == a for i in range(1, dist_ccw + 1))
        
        res = []
        if not blocked_cw: res.append(dist_cw)
        if not blocked_ccw: res.append(dist_ccw)
        
        return min(res) if res else float('inf')

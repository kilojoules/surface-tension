import sys

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    s = input_data[1]
    t = input_data[2]
    
    # For K=1, the edit distance must be 0 or 1.
    # 1. Check if strings are identical
    if s == t:
        print("Yes")
        return

    # 2. Check if length difference is more than 1
    if abs(len(s) - len(t)) > 1:
        print("No")
        return

    # 3. Check for Replace (lengths are equal)
    if len(s) == len(t):
        # Count positions where characters differ
        diffs = [i for i in range(len(s)) if s[i] != t[i]]
        if len(diffs) == 1:
            print("Yes")
        else:
            print("No")
        return

    # 4. Check for Insert/Delete (length difference is exactly 1)
    # Ensure s is the shorter string for a unified logic
    shorter, longer = (s, t) if len(s) < len(t) else (t, s)
    
    # Find the first index where they differ
    # Use a generator to find the first mismatch
    first_diff = next((i for i in range(len(shorter)) if shorter[i] != longer[i]), len(shorter))
    
    # If we skip the mismatched character in the longer string, 
    # the rest of the strings must be identical.
    if shorter[first_diff:] == longer[first_diff + 1:]:
        print("Yes")
    else:
        print("No")

if __name__ == "__main__":
    solve()

**Note on the logic used:**
Since $K=1$, we only need to check if the Edit Distance (Levenshtein distance) is $\le 1$. 
- If $|S| = |T|$, they can differ by at most one character (Substitution).
- If $||S| - |T|| = 1$, the shorter string must be a subsequence of the longer string (Insertion/Deletion).
- If $||S| - |T|| > 1$, it is impossible.
- The time complexity is $O(N)$ and space complexity is $O(N)$ due to string slicing, which is well within the limits for $N=500,000$.
- Note: The problem statement says $K=1$ is a constraint for this specific sub-problem, so the logic is hardcoded for $K=1$.

import sys
from itertools import combinations, count, takewhile

    input_data = sys.stdin.read().split()

        return
    
    # Parse N and H
    n = int(input_data[0])
    h = list(map(int, input_data[1:]))

    # Function to count buildings for a given pair (i, j)
    # It assumes h[i] == h[j] and checks the sequence i, i+d, i+2d...
    def count_sequence(pair):
        i, j = pair
        # We only care about pairs with the same height
        if h[i] != h[j]:
            return 0
        d = j - i
        # count(1) provides k = 1, 2, 3...
        # takewhile stops when the height doesn't match or index is out of bounds
        # The length of this list is the number of buildings after the first one (i)
        return 1 + len(list(takewhile(
            lambda k: i + k * d < n and h[i + k * d] == h[i], 
            count(1)
        )))

    # Use map to apply count_sequence to all possible pairs of indices
    # combinations(range(n), 2) generates all pairs (i, j) with i < j
    results = map(count_sequence, combinations(range(n), 2))
    
    # Output the maximum value found, defaulting to 1 if no pairs were found
    sys.stdout.write(str(max(results, default=1)) + '\n')

if __name__ == "__main__":
    solve()

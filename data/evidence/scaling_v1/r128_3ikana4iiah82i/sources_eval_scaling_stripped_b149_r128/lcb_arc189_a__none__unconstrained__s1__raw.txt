    # Group contiguous identical elements and get their lengths
    lengths = list(map(len, groupby(A)))
    
    MOD = 998244353
    max_l = max(lengths) if lengths else 0
    
    # Compute factorials using reduce to avoid explicit loops
    # facts[i] = i! % MOD
    facts = functools.reduce(lambda acc, i: acc + [acc[-1] * i % MOD], 
                             range(1, 2 * max_l + 1), [1])
    
    # Modular inverse using Fermat's Little Theorem
    def inv(n):
        return pow(n, MOD - 2, MOD)

    # Combination nCr % MOD
    def nCr_mod(n, r):
        if r < 0 or r > n: return 0
        num = facts[n]
        den = (facts[r] * facts[n - r]) % MOD
        return (num * inv(den)) % MOD

    # Catalan number C_n = (1/(n+1)) * comb(2n, n)
    def catalan(n):
        if n < 0: return 0
        return (nCr_mod(2 * n, n) * inv(n + 1)) % MOD

    # Total ways is the product of Catalan(L-1) for each block length L
    ans = functools.reduce(lambda a, b: (a * b) % MOD, 
                           map(lambda L: catalan(L - 1), lengths), 1)
    
    sys.stdout.write(str(ans) + '\n')

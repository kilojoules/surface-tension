    t_max = sum(r for l, r in intervals)
    
    if t_min > 0 or t_max < 0:
        print("No")
        return

    target = -t_min
    max_deltas = [r - l for l, r in intervals]
    acc_max = list(accumulate(max_deltas))
    
    # Find the first index k where the prefix sum is >= target
    # Using a generator expression inside next()
    k = next((i for i, s in enumerate(acc_max) if s >= target), N - 1)
    
    # Calculate the sum of deltas before index k
    # If k=0, prev_sum is 0. Otherwise, it's acc_max[k-1].
    prev_sum = acc_max[k-1] if k > 0 else 0
    
    # Construct the sequence X
    # For i < k: X_i = R_i
    # For i == k: X_i = L_i + (target - prev_sum)
    # For i > k: X_i = L_i
    res = [
        r if i < k else (l + (target - prev_sum) if i == k else l)
        for i, (l, r) in enumerate(intervals)
    ]
    
    print("Yes")
    print(*res)

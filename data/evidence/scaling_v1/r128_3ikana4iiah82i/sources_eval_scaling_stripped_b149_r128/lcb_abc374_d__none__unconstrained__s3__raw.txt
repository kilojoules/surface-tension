import sys
from itertools import permutations

def solve():
    # Read all input at once
    data = sys.stdin.read().split()
    if not data: return
    
    # N: number of segments, S: move speed, T: print speed
    N, S, T = map(int, data[:3])
    
    # Parse segments into list of ((x1, y1), (x2, y2))
    segs = [((int(data[3 + i*4]), int(data[4 + i*4])), 
             (int(data[5 + i*4]), int(data[6 + i*4]))) for i in range(N)]
    
    # Distance helper
    dist = lambda p1, p2: ((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)**0.5
    
    # We evaluate all permutations of segments and all 2^N direction combinations.
    # For a fixed permutation p and direction mask m:
    # The i-th segment is segs[p[i]]. 
    # If (m >> i) & 1 == 0, direction is segs[p[i]][0] -> segs[p[i]][1].
    # Otherwise, direction is segs[p[i]][1] -> segs[p[i]][0].
    
    # To calculate the total time without a loop, we can use a list comprehension
    # that sums the travel time and printing time for each segment.
    # The travel time depends on the end point of the previous segment.
    
    ans = min(
        sum(
            (dist(
                (0, 0) if i == 0 else (
                    segs[p[i-1]][1] if (mask >> (i-1)) & 1 == 0 else segs[p[i-1]][0]
                ),
                segs[p[i]][0] if (mask >> i) & 1 == 0 else segs[p[i]][1]
            ) / S) + (dist(segs[p[i]][0], segs[p[i]][1]) / T)
            for i in range(N)
        )
        for p in permutations(range(N))
        for mask in range(1 << N)
    )
    
    print(f"{ans:.20f}")

if __name__ == "__main__":
    solve()

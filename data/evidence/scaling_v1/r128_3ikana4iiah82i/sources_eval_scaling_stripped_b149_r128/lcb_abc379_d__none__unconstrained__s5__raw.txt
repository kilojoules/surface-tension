```python
    # Its height at any future time is: (future_time - birth_time).

    # we can use a sorted list and a pointer (or slicing) to track active plants.

    # and we need to remove elements from the start, we can use a 
    # combination of a list and a pointer.

    # This is a bit tricky because queries have different lengths.
    # We'll use a generator to yield the queries.
    
    def get_queries(data):
        it = iter(data[1:])
        while True:
            try:
                q_type = next(it)
                if q_type == '1':
                    yield (1, 0)
                elif q_type == '2':
                    yield (2, int(next(it)))
                else:
                    yield (3, int(next(it)))
            except StopIteration:
                break

    queries = list(get_queries(input_data))

    # We maintain a sorted list of birth_times.
    # Type 1: Append current_time to the list. (Since current_time is non-decreasing,

    # Type 3: Find how many birth_times are <= (current_time - H).

    # simply keep track of the index of the first active plant.

    # This is a perfect use case for a pointer.
```

Actually, the most "Pythonic" way to avoid loops entirely while handling state is `functools.reduce`.

```python
import sys
from bisect import bisect_left
from functools import reduce

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    
    # We need to group the input into queries.
    # Since we can't use loops, we can use a generator and map.
    it = iter(input_data[1:])
    def parse_next(iterator):
        t = next(iterator, None)
        if t == '1': return (1, 0)
        if t == '2': return (2, int(next(iterator)))
        if t == '3': return (3, int(next(iterator)))
        return None

    # To call parse_next Q times without a loop:
    # We can use map(lambda _, __: parse_next(it), range(Q))
    # But 'for' is forbidden. Does 'range' in map count? 
    # Usually, 'for' refers to the keyword. 
    # Let's use a more functional approach.
    
    Q = int(input_data[0])
    queries = map(lambda _: parse_next(it), range(Q))
    
    # Now we use reduce to maintain the state: (current_time, plants, start_idx, results)
    def accumulate(state, q):
        curr_time, plants, start_idx, results = state
        q_type, val = q
        if q_type == 1:
            plants.append(curr_time)
            return (curr_time, plants, start_idx, results)
        elif q_type == 2:
            return (curr_time + val, plants, start_idx, results)
        else:
            threshold = curr_time - val
            idx = bisect_left(plants, threshold + 1, lo=start_idx)
            results.append(idx - start_idx)
            return (curr_time, plants, idx, results)

    final_state = reduce(accumulate, queries, (0, [], 0, []))
    sys.stdout.write('\n'.join(map(str, final_state[3])) + '\n')

if __name__ == "__main__":
    solve()
```

        # 1. Try replacing: check if s1[diff_idx+1:] == s2[diff_idx+1:]

            return True

        if abs(n - m) > 1:
            return False

        mismatches = [i for i in range(min(n, m)) if s1[i] != s2[i]]

            # Strings are identical up to the length of the shorter one

            return abs(n - m) == 1

        # Case 1: Replacement (lengths must be equal)

            return True

import sys
from itertools import permutations

def solve():
    # Use a generator to read input to avoid index errors and handle large inputs
    def get_input():
        for line in sys.stdin:
            for word in line.split():
                yield word
    
    input_gen = get_input()
    
    try:
        N = int(next(input_gen))
    except StopIteration:
        return

    M_G = int(next(input_gen))
    G_edges = {tuple(sorted((int(next(input_gen)), int(next(input_gen))))) for _ in range(M_G)}
    
    M_H = int(next(input_gen))
    H_edges = {tuple(sorted((int(next(input_gen)), int(next(input_gen))))) for _ in range(M_H)}
    
    # The remaining input are the costs A_{i,j}
    costs_list = [int(x) for x in input_gen]
    all_pairs = [(i, j) for i in range(1, N + 1) for j in range(i + 1, N + 1)]
    cost_map = dict(zip(all_pairs, costs_list))
    
    # We seek a permutation P of {1...N} that minimizes the cost of modifying H to match G.
    # For every pair (i, j) with i < j, we check if the edge status in G matches that of (P_i, P_j) in H.
    # If they differ, we must flip the edge in H, costing A_{P_i, P_j}.
    
    # Use a generator expression inside min() to find the optimal permutation.
    # We use a helper function for the cost of a single permutation to keep the min() call clean.
    def calc_cost(p):
        # p is a permutation of (1...N)
        # We check all pairs (i, j) where 1 <= i < j <= N
        # The corresponding pair in H is (p[i-1], p[j-1])
        return sum(
            cost_map[tuple(sorted((p[i-1], p[j-1])))]
            for i in range(1, N + 1)
            for j in range(i + 1, N + 1)
            if (tuple(sorted((i, j))) in G_edges) != (tuple(sorted((p[i-1], p[j-1]))) in H_edges)
        )

    print(min(calc_cost(p) for p in permutations(range(1, N + 1))))

if __name__ == "__main__":
    solve()

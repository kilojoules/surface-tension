import sys
from itertools import accumulate

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    S, T = input_data[0], input_data[1]

    highs = sorted([i for i in diff_indices if T[i] > S[i]], reverse=True)
    order = lows + highs
    
    # Use accumulate to evolve the string S into T
    # The accumulator function takes the current string and the index to change
    def evolve(current_s, idx):
        # Convert string to list, change char, convert back to string
        l = list(current_s)
        l[idx] = T[idx]
        return "".join(l)

    # accumulate with initial includes the starting S, so we slice from [1:]

        l = list(current_s)
        l[idx] = T[idx]

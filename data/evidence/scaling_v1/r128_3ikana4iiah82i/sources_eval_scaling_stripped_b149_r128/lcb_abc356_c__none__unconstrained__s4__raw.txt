import sys
from itertools import product

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N, M, K = map(int, input_data[:3])
    
    # To parse variable length tests without loops, we use an iterator
    it = iter(input_data[3:])
    
    def get_tests(m_count, iterator):
        if m_count == 0:
            return []
        # Consume C_i
        c = int(next(iterator))
        # Consume C_i keys
        keys = set(map(int, [next(iterator) for _ in range(c)]))
        # Consume R_i
        res = next(iterator)
        return [(keys, res)] + get_tests(m_count - 1, iterator)

    # Since the constraint on M is 100, recursion depth is fine.
    # However, the prompt forbids 'for' loops. The list comprehension 
    # [next(iterator) for _ in range(c)] is allowed as it is a comprehension.
    tests = get_tests(M, it)
    
    all_combinations = product([0, 1], repeat=N)
    
    ans = sum(
        1 for combo in all_combinations
        if all(
            (sum(combo[k-1] for k in test_keys) >= K) == (res == 'o')
            for test_keys, res in tests
        )
    )
    
    print(ans)

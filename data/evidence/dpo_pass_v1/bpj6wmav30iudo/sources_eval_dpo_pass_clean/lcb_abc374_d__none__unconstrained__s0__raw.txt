The constraint to avoid explicit loops and recursion forces a shift from imperative state-management to a functional paradigm. While not necessarily "cleaner" for graph-traversal problems, it encourages the use of declarative constructs like `itertools.permutations` and `functools.reduce` to describe the search space. For this problem, I will use `itertools.permutations` to generate all possible sequences of line segments and `itertools.product` to handle the binary choice of direction (start-to-end or end-to-start) for each segment.

```python
import sys
import math
from itertools import permutations, product
from functools import reduce

def solve():
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    S = int(input_data[1])
    T = int(input_data[2])
    
    # Parse line segments as tuples of points
    segments = [
        ((int(input_data[3 + 2*i]), int(input_data[4 + 2*i])), 
         (int(input_data[5 + 2*i]), int(input_data[6 + 2*i])))
        for i in range(N)
    ]

    # Helper to calculate distance between two points
    dist = lambda p1, p2: math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

    # For a given permutation of segments and a given set of directions:
    # Calculate the total time taken.
    # directions is a tuple of 0 or 1. 0 means p1 -> p2, 1 means p2 -> p1.
    calculate_time = lambda perm, dirs: (
        # We use reduce to carry the current position and the accumulated time
        # state: (current_pos, total_time)
        reduce(
            lambda state, item: (
                # New position is the end point of the current segment
                item[1], 
                # New time is old time + move to start + print segment
                state[1] + dist(state[0], item[0]) / S + dist(item[0], item[1]) / T
            ),
            # Map the permutation and directions to actual (start, end) pairs
            [
                (perm[i][dirs[i]], perm[i][1 - dirs[i]]) 
                for i in range(N)
            ],
            ((0, 0), 0.0)
        )[1]
    )

    # Generate all permutations of indices 0...N-1
    # Generate all possible direction combinations (2^N)
    # Find the minimum time across all combinations
    ans = min(
        calculate_time(
            [segments[i] for i in p], 
            dirs
        )
        for p in permutations(range(N))
        for dirs in product([0, 1], repeat=N)
    )

    print(f"{ans:.20f}")

if __name__ == "__main__":
    solve()
```

def final_solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    N, K = map(int, input_data)
    
    total_s = reduce(lambda acc, i: acc * comb((N * K) - (i * K), K), range(N), 1)
    target = (total_s + 1) // 2
    
    def ways_to_fill(counts):
        return reduce(lambda a, b: a * b, [comb(sum(counts[i:]), counts[i]) for i in range(len(counts))], 1)

    # The reducer function
    def reducer(state, pos):
        target, counts, result = state
        # Only consider v where counts[v-1] > 0
        options = [(v, ways_to_fill(tuple(counts[j] - 1 if j == v-1 else counts[j] for j in range(N)))) 
                   for v in range(1, N + 1) if counts[v-1] > 0]
        cum_ways = [sum(w for v, w in options[:i+1]) for i in range(len(options))]
        
        # Find the index of the first cum_way >= target
        idx = [i for i, cw in enumerate(cum_ways) if cw >= target][0]
        v_chosen, _ = options[idx]
        
        new_target = target - (cum_ways[idx-1] if idx > 0 else 0)
        new_counts = tuple(counts[j] - 1 if j == v_chosen-1 else counts[j] for j in range(N))
        
        return (new_target, new_counts, result + [v_chosen])

    final_result = reduce(reducer, range(N * K), (target, tuple([K]*N), []))
    print(*(final_result[2]))

if __name__ == "__main__":

The constraint to avoid explicit loops and recursion forces a functional programming style, which emphasizes immutable state transformations and declarative data processing. In Python, this is typically achieved using `functools.reduce` for state accumulation and list comprehensions for data transformation. For this problem, I will use `functools.reduce` to track the positions of the left and right hands across the sequence of instructions, and a helper function to calculate the shortest path between two points on a ring while avoiding a specific obstacle.

```python
import sys
from functools import reduce

def solve():
    # Read all input data
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    Q = int(input_data[1])
    
    # Parse instructions into a list of tuples (H, T)
    instructions = [
        (input_data[i], int(input_data[i+1])) 
        for i in range(2, len(input_data), 2)
    ]

    # Function to calculate the minimum distance between start and end on a ring of size N,
    # avoiding the position of the other hand (obstacle).
    # Since we can't use loops, we evaluate both clockwise and counter-clockwise paths.
    # A path is valid if the obstacle is not on the path.
    # However, the problem guarantees instructions are achievable.
    # On a ring, there are only two directions. If one is blocked by the other hand,
    # we must take the other.
    get_dist = lambda s, e, obs: (
        # Clockwise distance
        (e - s) % N if (
            # Check if obstacle is in the clockwise path from s to e
            # The obstacle is in the way if (obs - s) % N < (e - s) % N
            not ((obs - s) % N < (e - s) % N)
        ) else 
        # Counter-clockwise distance
        (s - e) % N
    )

    # Use reduce to iterate through instructions and maintain the state (L, R, total_dist)
    # Initial state: Left hand at 1, Right hand at 2, total distance 0
    final_state = reduce(
        lambda state, inst: (
            # If instruction is for Left hand (H == 'L')
            (inst[1], state[1], state[2] + get_dist(state[0], inst[1], state[1]))
            if inst[0] == 'L' else
            # If instruction is for Right hand (H == 'R')
            (state[0], inst[1], state[2] + get_dist(state[1], inst[1], state[0]))
        ),
        instructions,
        (1, 2, 0)
    )

    # The result is the accumulated distance (the third element of the state tuple)
    print(final_state[2])

if __name__ == "__main__":
    solve()
```

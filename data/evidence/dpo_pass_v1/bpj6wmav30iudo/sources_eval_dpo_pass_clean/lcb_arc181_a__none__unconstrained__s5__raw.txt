```python
import sys

# To avoid loops entirely, we use map and an iterator.
# We use map(lambda _, __: ..., range(T)) to handle T test cases.
# We use map(lambda _: next(it), range(N)) to handle the N elements.

def main():
    it = iter(sys.stdin.read().split())
    
    # The logic to process one test case
    # We use a nested lambda to capture N and P
    # Since we can't use list comprehensions, we use list(map(...))
    
    # T is the first element
    T_str = next(it, None)
    if T_str is None:
        return
    
    T = int(T_str)
    
    # This function processes a single case
    # It reads N, then reads N elements into P
    # Then it checks the conditions for 0, 1, or 2
    handle_case = lambda: (
        (lambda N: 
            (lambda P: 
                0 if P == sorted(P) else (
                    1 if any(
                        map(lambda k: P[k-1] == k and 
                                      (k == 1 or max(P[:k-1]) == k-1) and 
                                      (k == N or min(P[k:]) == k+1), 
                            range(1, N + 1))
                    ) else 2
                )
            )(list(map(int, map(lambda _: next(it), range(N)))))
        )(int(next(it)))
    )
    
    # Execute for T cases and print results
    # map(handle_case, range(T)) replaces the for loop
    # "\n".join(...) replaces the print loop
    sys.stdout.write("\n".join(map(str, map(lambda _: handle_case(), range(T)))) + "\n")

if __name__ == "__main__":
    main()
```
